    97→  const m = Math.floor(seconds / 60)
    98→  const s = seconds % 60
    99→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`

   352→  // --- Test Functions ---
   353→  async function startTest(test: LocalTest) {
   354→    const fullTest = await fetchTestById(test.id)
   355→    if (!fullTest) return
   356→    setSelectedTest(fullTest)
   357→    setAnswers({})

   364→
   365→  async function handleFinishTest() {
   366→    if (timerRef.current) clearInterval(timerRef.current)
   367→    setTestActive(false)
   368→
   369→    if (!selectedTest) return
   370→
   371→    const test = selectedTest
   372→    const totalQuestions = test.questions.length
   373→    let correctCount = 0
   374→    let wrongCount = 0
   375→    let skippedCount = 0
   376→
   377→    test.questions.forEach(q => {
   378→      const userAnswer = answers[q.id]
   379→      if (!userAnswer) {
   380→        skippedCount++
   381→      } else if (userAnswer === q.correctAnswer) {
   382→        correctCount++
   383→      } else {
   384→        wrongCount++
   385→      }
   386→    })
   387→
   388→    const score = correctCount * test.markingCorrect - wrongCount * Math.abs(test.markingWrong)
   389→    const maxScore = totalQuestions * test.markingCorrect
   390→    const timeTaken = test.duration * 60 - timeLeft
   391→
   392→    const result = await storeResult({
   393→      testId: test.id,
   394→      testName: test.title,
   395→      examName: test.exam.name,
   396→      userId: auth.getUserId() || 'anonymous',
   397→      correctCount,
   398→      wrongCount,
   399→      skippedCount,
   400→      score,
   401→      maxScore,
   402→      timeTaken,
   403→      totalQuestions,
   404→      answers,
   405→    })
   406→
   407→    setLastResult(result)
   408→    navigateTo('results')
   409→  }

   432→  // --- Stats from localStorage ---
   433→  function getUserStats() {
   434→    try {
   435→      const data = localStorage.getItem('examprep_results')
   436→      if (!data) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   437→      const results: TestResult[] = JSON.parse(data)
   438→      const userId = auth.getUserId()
   439→      const userResults = userId ? results.filter((r: TestResult) => r.userId === userId) : results
   440→      const testsTaken = userResults.length
   441→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.score / r.maxScore) * 100, 0) / testsTaken) : 0
   442→      const bestRank = 1 // simplified
   443→      return { testsTaken, avgScore, bestRank }
   444→    } catch {
   445→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   446→    }
   447→  }
