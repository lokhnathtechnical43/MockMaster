   423→  // --- Test Functions ---
   424→  async function startTest(test: LocalTest) {
   425→    const fullTest = await fetchTestById(test.id)
   426→    if (!fullTest) return
   427→    setSelectedTest(fullTest)
   428→    setAnswers({})
   429→    setMarkedForReview(new Set())
   430→    setCurrentQuestionIndex(0)
   431→    setTimeLeft(fullTest.duration * 60)
   432→    setTestActive(true)
   433→    navigateTo('test-taking')
   434→  }
   435→
   436→  async function handleFinishTest() {
   437→    if (timerRef.current) clearInterval(timerRef.current)
   438→    setTestActive(false)
   439→
   440→    if (!selectedTest) return
   441→
   442→    const test = selectedTest
   443→    const totalQuestions = test.questions.length
   444→    let correctCount = 0
   445→    let wrongCount = 0
   446→    let skippedCount = 0
   447→
   448→    test.questions.forEach(q => {
   449→      const userAnswer = answers[q.id]
   450→      if (!userAnswer) {
   451→        skippedCount++
   452→      } else if (userAnswer === q.correctAnswer) {
   453→        correctCount++
   454→      } else {
   455→        wrongCount++
   456→      }
   457→    })
   458→
   459→    const score = correctCount * test.markingCorrect - wrongCount * Math.abs(test.markingWrong)
   460→    const maxScore = totalQuestions * test.markingCorrect
   461→    const timeTaken = test.duration * 60 - timeLeft
   462→
   463→    const result = await storeResult({
   464→      testId: test.id,
   465→      testName: test.title,
   466→      examName: test.exam.name,
   467→      userId: auth.getUserId() || 'anonymous',
   468→      correctCount,
   469→      wrongCount,
   470→      skippedCount,
   471→      score,
   472→      maxScore,
   473→      timeTaken,
   474→      totalQuestions,
   475→      answers,
   476→    })
   477→
   478→    setLastResult(result)
   479→    navigateTo('results')
   480→  }

   503→  // --- Stats from localStorage ---
   504→  function getUserStats() {
   505→    try {
   506→      const data = localStorage.getItem('examprep_results')
   507→      if (!data) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   508→      const results: TestResult[] = JSON.parse(data)
   509→      const userId = auth.getUserId()
   510→      const userResults = userId ? results.filter((r: TestResult) => r.userId === userId) : results
   511→      const testsTaken = userResults.length
   512→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.score / r.maxScore) * 100, 0) / testsTaken) : 0
   513→      const bestRank = 1 // simplified
   514→      return { testsTaken, avgScore, bestRank }
   515→    } catch {
   516→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   517→    }
   518→  }
