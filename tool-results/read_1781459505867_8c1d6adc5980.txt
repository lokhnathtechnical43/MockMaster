     1→     1→'use client'
     2→     2→
     3→     3→import { useState, useEffect, useCallback } from 'react'
     4→     4→import { Button } from '@/components/ui/button'
     5→     5→import { Card, CardContent } from '@/components/ui/card'
     6→     6→import { Badge } from '@/components/ui/badge'
     7→     7→import { Input } from '@/components/ui/input'
     8→     8→import { Textarea } from '@/components/ui/textarea'
     9→     9→import {
    10→    10→  BookOpen, ChevronRight, Plus, Trash2, Edit3, Save,
    11→    11→  ArrowLeft, RefreshCw, ListChecks, HelpCircle, Database
    12→    12→} from 'lucide-react'
    13→    13→import {
    14→    14→  getCategories, addCategory, updateCategory, deleteCategory,
    15→    15→  getExams, addExam, updateExam, deleteExam,
    16→    16→  getTests, addTest, updateTest, deleteTest,
    17→    17→  getQuestions, addQuestion, updateQuestion, deleteQuestion,
    18→    18→  addBatchQuestions, seedFirestoreIfEmpty,
    19→    19→  type FirestoreExamCategory, type FirestoreExam, type FirestoreTest, type FirestoreQuestion
    20→    20→} from '@/lib/firestore-service'
    21→    21→import { getCategories as getLocalCategories } from '@/lib/local-data'
    22→    22→
    23→    23→type ViewLevel = 'categories' | 'exams' | 'tests' | 'questions'
    24→    24→
    25→    25→export default function ExamsTab() {
    26→    26→  const [viewLevel, setViewLevel] = useState<ViewLevel>('categories')
    27→    27→  const [loading, setLoading] = useState(true)
    28→    28→  const [seeding, setSeeding] = useState(false)
    29→    29→
    30→    30→  // Categories
    31→    31→  const [categories, setCategories] = useState<FirestoreExamCategory[]>([])
    32→    32→  const [selectedCategoryId, setSelectedCategoryId] = useState<string | null>(null)
    33→    33→  const [selectedCategoryName, setSelectedCategoryName] = useState('')
    34→    34→
    35→    35→  // Exams
    36→    36→  const [exams, setExams] = useState<FirestoreExam[]>([])
    37→    37→  const [selectedExamId, setSelectedExamId] = useState<string | null>(null)
    38→    38→  const [selectedExamName, setSelectedExamName] = useState('')
    39→    39→
    40→    40→  // Tests
    41→    41→  const [tests, setTests] = useState<FirestoreTest[]>([])
    42→    42→  const [selectedTestId, setSelectedTestId] = useState<string | null>(null)
    43→    43→  const [selectedTestName, setSelectedTestName] = useState('')
    44→    44→
    45→    45→  // Questions
    46→    46→  const [questions, setQuestions] = useState<FirestoreQuestion[]>([])
    47→    47→
    48→    48→  // Edit states
    49→    49→  const [editingId, setEditingId] = useState<string | null>(null)
    50→    50→  const [editName, setEditName] = useState('')
    51→    51→  const [editDesc, setEditDesc] = useState('')
    52→    52→
    53→    53→  // Add states
    54→    54→  const [showAdd, setShowAdd] = useState(false)
    55→    55→  const [addName, setAddName] = useState('')
    56→    56→  const [addDesc, setAddDesc] = useState('')
    57→    57→  const [addSlug, setAddSlug] = useState('')
    58→    58→
    59→    59→  // Add question
    60→    60→  const [showAddQuestion, setShowAddQuestion] = useState(false)
    61→    61→  const [qText, setQText] = useState('')
    62→    62→  const [qA, setQA] = useState('')
    63→    63→  const [qB, setQB] = useState('')
    64→    64→  const [qC, setQC] = useState('')
    65→    65→  const [qD, setQD] = useState('')
    66→    66→  const [qAns, setQAns] = useState('A')
    67→    67→  const [qExplanation, setQExplanation] = useState('')
    68→    68→  const [qSubject, setQSubject] = useState('')
    69→    69→
    70→    70→  // Batch questions
    71→    71→  const [showBatch, setShowBatch] = useState(false)
    72→    72→  const [batchText, setBatchText] = useState('')
    73→    73→
    74→    74→  const loadCategories = useCallback(async () => {
    75→    75→    setLoading(true)
    76→    76→    try {
    77→    77→      const cats = await getCategories()
    78→    78→      setCategories(cats.map(c => ({
    79→    79→        id: c.id, name: c.name, slug: c.slug, icon: c.icon,
    80→    80→        description: c.description, order: c.order
    81→    81→      })))
    82→    82→    } catch {
    83→    83→      // Fallback to local
    84→    84→      const local = getLocalCategories()
    85→    85→      setCategories(local.map(c => ({
    86→    86→        id: c.id, name: c.name, slug: c.slug, icon: c.icon,
    87→    87→        description: c.description, order: c.order
    88→    88→      })))
    89→    89→    } finally {
    90→    90→      setLoading(false)
    91→    91→    }
    92→    92→  }, [])
    93→    93→
    94→    94→  const loadExams = useCallback(async (categoryId: string) => {
    95→    95→    setLoading(true)
    96→    96→    try {
    97→    97→      const data = await getExams(categoryId)
    98→    98→      setExams(data)
    99→    99→    } catch { setExams([]) } finally { setLoading(false) }
   100→   100→  }, [])
   101→   101→
   102→   102→  const loadTests = useCallback(async (examId: string) => {
   103→   103→    setLoading(true)
   104→   104→    try {
   105→   105→      const data = await getTests(examId)
   106→   106→      setTests(data)
   107→   107→    } catch { setTests([]) } finally { setLoading(false) }
   108→   108→  }, [])
   109→   109→
   110→   110→  const loadQuestions = useCallback(async (testId: string) => {
   111→   111→    setLoading(true)
   112→   112→    try {
   113→   113→      const data = await getQuestions(testId)
   114→   114→      setQuestions(data)
   115→   115→    } catch { setQuestions([]) } finally { setLoading(false) }
   116→   116→  }, [])
   117→   117→
   118→   118→  useEffect(() => { loadCategories() }, [loadCategories])
   119→   119→
   120→   120→  const handleSelectCategory = (cat: FirestoreExamCategory) => {
   121→   121→    setSelectedCategoryId(cat.id)
   122→   122→    setSelectedCategoryName(cat.name)
   123→   123→    loadExams(cat.id)
   124→   124→    setViewLevel('exams')
   125→   125→  }
   126→   126→
   127→   127→  const handleSelectExam = (exam: FirestoreExam) => {
   128→   128→    setSelectedExamId(exam.id)
   129→   129→    setSelectedExamName(exam.name)
   130→   130→    loadTests(exam.id)
   131→   131→    setViewLevel('tests')
   132→   132→  }
   133→   133→
   134→   134→  const handleSelectTest = (test: FirestoreTest) => {
   135→   135→    setSelectedTestId(test.id)
   136→   136→    setSelectedTestName(test.title)
   137→   137→    loadQuestions(test.id)
   138→   138→    setViewLevel('questions')
   139→   139→  }
   140→   140→
   141→   141→  const handleBack = () => {
   142→   142→    if (viewLevel === 'questions') { setViewLevel('tests'); setQuestions([]) }
   143→   143→    else if (viewLevel === 'tests') { setViewLevel('exams'); setTests([]) }
   144→   144→    else if (viewLevel === 'exams') { setViewLevel('categories'); setExams([]) }
   145→   145→    setShowAdd(false)
   146→   146→    setEditingId(null)
   147→   147→  }
   148→   148→
   149→   149→  // --- CRUD Handlers ---
   150→   150→  const handleAddCategory = async () => {
   151→   151→    if (!addName.trim()) return
   152→   152→    try {
   153→   153→      await addCategory({
   154→   154→        name: addName, slug: addSlug || addName.toLowerCase().replace(/\s+/g, '-'),
   155→   155→        icon: '📋', description: addDesc, order: categories.length + 1
   156→   156→      })
   157→   157→      setAddName(''); setAddDesc(''); setAddSlug(''); setShowAdd(false)
   158→   158→      loadCategories()
   159→   159→    } catch { alert('Failed to add category') }
   160→   160→  }
   161→   161→
   162→   162→  const handleAddExam = async () => {
   163→   163→    if (!addName.trim() || !selectedCategoryId) return
   164→   164→    try {
   165→   165→      await addExam({
   166→   166→        name: addName, slug: addSlug || addName.toLowerCase().replace(/\s+/g, '-'),
   167→   167→        description: addDesc, totalQuestions: 0, duration: 60,
   168→   168→        markingScheme: '1/-0.25', order: exams.length + 1,
   169→   169→        testCount: 0, categoryId: selectedCategoryId
   170→   170→      })
   171→   171→      setAddName(''); setAddDesc(''); setAddSlug(''); setShowAdd(false)
   172→   172→      loadExams(selectedCategoryId)
   173→   173→    } catch { alert('Failed to add exam') }
   174→   174→  }
   175→   175→
   176→   176→  const handleAddTest = async () => {
   177→   177→    if (!addName.trim() || !selectedExamId) return
   178→   178→    try {
   179→   179→      await addTest({
   180→   180→        title: addName, slug: addSlug || addName.toLowerCase().replace(/\s+/g, '-'),
   181→   181→        description: addDesc, totalQuestions: 0, duration: 30,
   182→   182→        markingCorrect: 1, markingWrong: -0.25, markingSkipped: 0,
   183→   183→        difficulty: 'medium', isFree: true, isLive: true,
   184→   184→        examId: selectedExamId, examName: selectedExamName,
   185→   185→        examSlug: selectedExamName.toLowerCase().replace(/\s+/g, '-')
   186→   186→      })
   187→   187→      setAddName(''); setAddDesc(''); setAddSlug(''); setShowAdd(false)
   188→   188→      loadTests(selectedExamId)
   189→   189→    } catch { alert('Failed to add test') }
   190→   190→  }
   191→   191→
   192→   192→  const handleAddQuestion = async () => {
   193→   193→    if (!qText.trim() || !selectedTestId) return
   194→   194→    try {
   195→   195→      await addQuestion({
   196→   196→        questionText: qText, questionImage: null,
   197→   197→        optionA: qA, optionB: qB, optionC: qC, optionD: qD,
   198→   198→        correctAnswer: qAns, explanation: qExplanation || null,
   199→   199→        subject: qSubject || null, order: questions.length + 1,
   200→   200→        testId: selectedTestId
   201→   201→      })
   202→   202→      setQText(''); setQA(''); setQB(''); setQC(''); setQD('')
   203→   203→      setQAns('A'); setQExplanation(''); setQSubject('')
   204→   204→      setShowAddQuestion(false)
   205→   205→      loadQuestions(selectedTestId)
   206→   206→    } catch { alert('Failed to add question') }
   207→   207→  }
   208→   208→
   209→   209→  const handleBatchQuestions = async () => {
   210→   210→    if (!batchText.trim() || !selectedTestId) return
   211→   211→    try {
   212→   212→      const lines = batchText.trim().split('\n').filter(l => l.trim())
   213→   213→      const qs = lines.map((line, i) => {
   214→   214→        const parts = line.split('|')
   215→   215→        return {
   216→   216→          questionText: parts[0]?.trim() || '',
   217→   217→          optionA: parts[1]?.trim() || '',
   218→   218→          optionB: parts[2]?.trim() || '',
   219→   219→          optionC: parts[3]?.trim() || '',
   220→   220→          optionD: parts[4]?.trim() || '',
   221→   221→          correctAnswer: parts[5]?.trim() || 'A',
   222→   222→          explanation: parts[6]?.trim() || null,
   223→   223→          subject: parts[7]?.trim() || null,
   224→   224→          order: i + 1,
   225→   225→          testId: selectedTestId
   226→   226→        }
   227→   227→      }).filter(q => q.questionText)
   228→   228→      await addBatchQuestions(selectedTestId, qs)
   229→   229→      setBatchText(''); setShowBatch(false)
   230→   230→      loadQuestions(selectedTestId)
   231→   231→    } catch { alert('Failed to add batch questions') }
   232→   232→  }
   233→   233→
   234→   234→  const handleDelete = async (type: string, id: string) => {
   235→   235→    if (!confirm(`Delete this ${type}?`)) return
   236→   236→    try {
   237→   237→      if (type === 'category') { await deleteCategory(id); loadCategories() }
   238→   238→      else if (type === 'exam' && selectedCategoryId) { await deleteExam(id); loadExams(selectedCategoryId) }
   239→   239→      else if (type === 'test' && selectedExamId) { await deleteTest(id); loadTests(selectedExamId) }
   240→   240→      else if (type === 'question' && selectedTestId) { await deleteQuestion(id); loadQuestions(selectedTestId) }
   241→   241→    } catch { alert(`Failed to delete ${type}`) }
   242→   242→  }
   243→   243→
   244→   244→  const handleSeed = async () => {
   245→   245→    if (!confirm('Seed Firestore from local data? This may take a moment.')) return
   246→   246→    setSeeding(true)
   247→   247→    try {
   248→   248→      await seedFirestoreIfEmpty()
   249→   249→      alert('Firestore seeded successfully!')
   250→   250→      loadCategories()
   251→   251→    } catch { alert('Failed to seed Firestore') }
   252→   252→    finally { setSeeding(false) }
   253→   253→  }
   254→   254→
   255→   255→  const handleStartEdit = (id: string, name: string, desc: string) => {
   256→   256→    setEditingId(id)
   257→   257→    setEditName(name)
   258→   258→    setEditDesc(desc || '')
   259→   259→  }
   260→   260→
   261→   261→  const handleSaveEdit = async (type: string, id: string) => {
   262→   262→    try {
   263→   263→      if (type === 'category') {
   264→   264→        await updateCategory(id, { name: editName, description: editDesc })
   265→   265→        loadCategories()
   266→   266→      } else if (type === 'exam') {
   267→   267→        await updateExam(id, { name: editName, description: editDesc })
   268→   268→        if (selectedCategoryId) loadExams(selectedCategoryId)
   269→   269→      } else if (type === 'test') {
   270→   270→        await updateTest(id, { title: editName, description: editDesc })
   271→   271→        if (selectedExamId) loadTests(selectedExamId)
   272→   272→      }
   273→   273→      setEditingId(null)
   274→   274→    } catch { alert('Failed to update') }
   275→   275→  }
   276→   276→
   277→   277→  return (
   278→   278→    <div className="space-y-4">
   279→   279→      {/* Breadcrumb + Actions */}
   280→   280→      <div className="flex items-center justify-between">
   281→   281→        <div className="flex items-center gap-2 text-sm">
   282→   282→          {viewLevel !== 'categories' && (
   283→   283→            <button onClick={handleBack} className="text-violet-600 hover:text-violet-800 font-medium flex items-center gap-1">
   284→   284→              <ArrowLeft className="w-4 h-4" /> Back
   285→   285→            </button>
   286→   286→          )}
   287→   287→          <span className="text-gray-400">
   288→   288→            {viewLevel === 'categories' ? 'Categories' :
   289→   289→             viewLevel === 'exams' ? `› ${selectedCategoryName} › Exams` :
   290→   290→             viewLevel === 'tests' ? `› ${selectedCategoryName} › ${selectedExamName} › Tests` :
   291→   291→             `› ${selectedCategoryName} › ${selectedExamName} › ${selectedTestName} › Questions`}
   292→   292→          </span>
   293→   293→        </div>
   294→   294→        <div className="flex items-center gap-2">
   295→   295→          <Button
   296→   296→            variant="outline"
   297→   297→            size="sm"
   298→   298→            className="rounded-xl text-xs"
   299→   299→            onClick={handleSeed}
   300→   300→            disabled={seeding}
   301→   301→          >
   302→   302→            <Database className="w-3 h-3 mr-1" />
   303→   303→            {seeding ? 'Seeding...' : 'Seed Firestore'}
   304→   304→          </Button>
   305→   305→        </div>
   306→   306→      </div>
   307→   307→
   308→   308→      {/* Loading */}
   309→   309→      {loading && (
   310→   310→        <div className="flex items-center justify-center py-12">
   311→   311→          <RefreshCw className="w-6 h-6 text-gray-400 animate-spin" />
   312→   312→        </div>
   313→   313→      )}
   314→   314→
   315→   315→      {/* ===== CATEGORIES VIEW ===== */}
   316→   316→      {!loading && viewLevel === 'categories' && (
   317→   317→        <div className="space-y-4">
   318→   318→          <div className="flex items-center justify-between">
   319→   319→            <h3 className="font-bold text-sm">Categories ({categories.length})</h3>
   320→   320→            <Button size="sm" className="rounded-xl text-xs" onClick={() => { setShowAdd(!showAdd); setAddName(''); setAddDesc(''); setAddSlug('') }}>
   321→   321→              <Plus className="w-3 h-3 mr-1" /> Add Category
   322→   322→            </Button>
   323→   323→          </div>
   324→   324→
   325→   325→          {showAdd && (
   326→   326→            <Card className="border-0 shadow-md bg-orange-50/50">
   327→   327→              <CardContent className="p-4 space-y-2">
   328→   328→                <Input placeholder="Category name" value={addName} onChange={e => setAddName(e.target.value)} className="rounded-xl" />
   329→   329→                <Input placeholder="Slug (auto-generated if empty)" value={addSlug} onChange={e => setAddSlug(e.target.value)} className="rounded-xl" />
   330→   330→                <Input placeholder="Description" value={addDesc} onChange={e => setAddDesc(e.target.value)} className="rounded-xl" />
   331→   331→                <div className="flex gap-2">
   332→   332→                  <Button size="sm" className="rounded-xl" onClick={handleAddCategory} disabled={!addName.trim()}>Add</Button>
   333→   333→                  <Button size="sm" variant="outline" className="rounded-xl" onClick={() => setShowAdd(false)}>Cancel</Button>
   334→   334→                </div>
   335→   335→              </CardContent>
   336→   336→            </Card>
   337→   337→          )}
   338→   338→
   339→   339→          {categories.length === 0 ? (
   340→   340→            <Card className="border-0 shadow-sm">
   341→   341→              <CardContent className="p-8 text-center">
   342→   342→                <BookOpen className="w-12 h-12 text-gray-300 mx-auto mb-3" />
   343→   343→                <p className="text-gray-400 text-sm">No categories found</p>
   344→   344→              </CardContent>
   345→   345→            </Card>
   346→   346→          ) : (
   347→   347→            categories.map(cat => (
   348→   348→              <Card key={cat.id} className="border-0 shadow-sm hover:shadow-md transition-shadow">
   349→   349→                <CardContent className="p-3 flex items-center gap-3">
   350→   350→                  {editingId === cat.id ? (
   351→   351→                    <div className="flex-1 space-y-2">
   352→   352→                      <Input value={editName} onChange={e => setEditName(e.target.value)} className="rounded-xl text-sm" />
   353→   353→                      <Input value={editDesc} onChange={e => setEditDesc(e.target.value)} placeholder="Description" className="rounded-xl text-sm" />
   354→   354→                      <div className="flex gap-2">
   355→   355→                        <Button size="sm" className="rounded-xl" onClick={() => handleSaveEdit('category', cat.id)}>
   356→   356→                          <Save className="w-3 h-3 mr-1" /> Save
   357→   357→                        </Button>
   358→   358→                        <Button size="sm" variant="outline" className="rounded-xl" onClick={() => setEditingId(null)}>Cancel</Button>
   359→   359→                      </div>
   360→   360→                    </div>
   361→   361→                  ) : (
   362→   362→                    <>
   363→   363→                      <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-orange-100 to-orange-200 flex items-center justify-center flex-shrink-0 text-lg">
   364→   364→                        {cat.icon || '📋'}
   365→   365→                      </div>
   366→   366→                      <button className="flex-1 min-w-0 text-left" onClick={() => handleSelectCategory(cat)}>
   367→   367→                        <p className="font-semibold text-sm truncate">{cat.name}</p>
   368→   368→                        <p className="text-gray-400 text-xs truncate">{cat.description || cat.slug}</p>
   369→   369→                      </button>
   370→   370→                      <div className="flex items-center gap-1">
   371→   371→                        <button onClick={() => handleStartEdit(cat.id, cat.name, cat.description)} className="w-8 h-8 rounded-lg bg-blue-50 hover:bg-blue-100 flex items-center justify-center">
   372→   372→                          <Edit3 className="w-3.5 h-3.5 text-blue-500" />
   373→   373→                        </button>
   374→   374→                        <button onClick={() => handleDelete('category', cat.id)} className="w-8 h-8 rounded-lg bg-red-50 hover:bg-red-100 flex items-center justify-center">
   375→   375→                          <Trash2 className="w-3.5 h-3.5 text-red-400" />
   376→   376→                        </button>
   377→   377→                        <button onClick={() => handleSelectCategory(cat)} className="w-8 h-8 rounded-lg bg-gray-50 hover:bg-gray-100 flex items-center justify-center">
   378→   378→                          <ChevronRight className="w-4 h-4 text-gray-400" />
   379→   379→                        </button>
   380→   380→                      </div>
   381→   381→                    </>
   382→   382→                  )}
   383→   383→                </CardContent>
   384→   384→              </Card>
   385→   385→            ))
   386→   386→          )}
   387→   387→        </div>
   388→   388→      )}
   389→   389→
   390→   390→      {/* ===== EXAMS VIEW ===== */}
   391→   391→      {!loading && viewLevel === 'exams' && (
   392→   392→        <div className="space-y-4">
   393→   393→          <div className="flex items-center justify-between">
   394→   394→            <h3 className="font-bold text-sm">Exams in {selectedCategoryName} ({exams.length})</h3>
   395→   395→            <Button size="sm" className="rounded-xl text-xs" onClick={() => { setShowAdd(!showAdd); setAddName(''); setAddDesc(''); setAddSlug('') }}>
   396→   396→              <Plus className="w-3 h-3 mr-1" /> Add Exam
   397→   397→            </Button>
   398→   398→          </div>
   399→   399→
   400→   400→          {showAdd && (
   401→   401→            <Card className="border-0 shadow-md bg-orange-50/50">
   402→   402→              <CardContent className="p-4 space-y-2">
   403→   403→                <Input placeholder="Exam name" value={addName} onChange={e => setAddName(e.target.value)} className="rounded-xl" />
   404→   404→                <Input placeholder="Slug" value={addSlug} onChange={e => setAddSlug(e.target.value)} className="rounded-xl" />
   405→   405→                <Input placeholder="Description" value={addDesc} onChange={e => setAddDesc(e.target.value)} className="rounded-xl" />
   406→   406→                <div className="flex gap-2">
   407→   407→                  <Button size="sm" className="rounded-xl" onClick={handleAddExam} disabled={!addName.trim()}>Add</Button>
   408→   408→                  <Button size="sm" variant="outline" className="rounded-xl" onClick={() => setShowAdd(false)}>Cancel</Button>
   409→   409→                </div>
   410→   410→              </CardContent>
   411→   411→            </Card>
   412→   412→          )}
   413→   413→
