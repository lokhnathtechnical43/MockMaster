    99→function formatTime(seconds: number): string {
   100→  const m = Math.floor(seconds / 60)
   101→  const s = seconds % 60
   102→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
   103→}

   480→  // --- Test Functions ---
   481→  async function startTest(test: LocalTest) {
   482→    const fullTest = await fetchTestById(test.id)
   483→    if (!fullTest) return
   484→    setSelectedTest(fullTest)
   485→    setAnswers({})
   486→    setMarkedForReview(new Set())
   487→    setCurrentQuestionIndex(0)
   488→    setTimeLeft(fullTest.duration * 60)
   489→    setTestActive(true)
   490→    navigateTo('test-taking')
   491→  }
   492→
   493→  async function handleFinishTest() {
   494→    if (timerRef.current) clearInterval(timerRef.current)
   495→    setTestActive(false)
   496→
   497→    if (!selectedTest) return
   498→
   499→    const test = selectedTest
   500→    const totalQuestions = test.questions.length
   501→    let correctCount = 0
   502→    let wrongCount = 0
   503→    let skippedCount = 0
   504→
   505→    test.questions.forEach(q => {
   506→      const userAnswer = answers[q.id]
   507→      if (!userAnswer) {
   508→        skippedCount++
   509→      } else if (userAnswer === q.correctAnswer) {
   510→        correctCount++
   511→      } else {
   512→        wrongCount++
   513→      }
   514→    })
   515→
   516→    const score = correctCount * test.markingCorrect - wrongCount * Math.abs(test.markingWrong)
   517→    const maxScore = totalQuestions * test.markingCorrect
   518→    const timeTaken = test.duration * 60 - timeLeft
   519→
   520→    const result = await storeResult({
   521→      testId: test.id,
   522→      testName: test.title,
   523→      examName: test.exam.name,
   524→      userId: auth.getUserId() || 'anonymous',
   525→      correctCount,
   526→      wrongCount,
   527→      skippedCount,
   528→      score,
   529→      maxScore,
   530→      timeTaken,
   531→      totalQuestions,
   532→      answers,
   533→    })
   534→
   535→    setLastResult(result)
   536→    navigateTo('results')
   537→  }

   560→  // --- Stats from localStorage ---
   561→  function getUserStats() {
   562→    try {
   563→      const data = localStorage.getItem('examprep_results')
   564→      if (!data) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   565→      const results: TestResult[] = JSON.parse(data)
   566→      const userId = auth.getUserId()
   567→      const userResults = userId ? results.filter((r: TestResult) => r.userId === userId) : results
   568→      const testsTaken = userResults.length
   569→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.score / r.maxScore) * 100, 0) / testsTaken) : 0
   570→      const bestRank = 1 // simplified
   571→      return { testsTaken, avgScore, bestRank }
   572→    } catch {
   573→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   574→    }
   575→  }
