   221→   221→// ============================================================
   222→   222→// Helper: Safe Firestore call with fallback
   223→   223→// ============================================================
   224→   224→
   225→   225→async function firestoreOperation<T>(
   226→   226→  firestoreFn: () => Promise<T>,
   227→   227→  fallbackFn: () => T
   228→   228→): Promise<T> {
   229→   229→  if (!useFirestore || !isFirebaseReady() || !db) {
   230→   230→    return fallbackFn()
   231→   231→  }
   232→   232→  try {
   233→   233→    return await firestoreFn()
   234→   234→  } catch (error) {
   235→   235→    console.error('[Firestore] Operation failed, falling back to local data:', error)
   236→   236→    return fallbackFn()
   237→   237→  }
   238→   238→}

   244→   244→/**
   245→   245→ * Get all categories with their nested exams.
   246→   246→ * Firestore: fetches categories & exams separately, then merges.
   247→   247→ * Fallback: returns local-data categories.
   248→   248→ */
   249→   249→export async function getCategories(): Promise<LocalExamCategory[]> {
   250→   250→  return firestoreOperation(
   251→   251→    async () => {
   252→   252→      const catSnap = await getDocs(
   253→   253→        query(collection(db, COLLECTIONS.categories), orderBy('order'))
   254→   254→      )
   255→   255→      const examSnap = await getDocs(
   256→   256→        query(collection(db, COLLECTIONS.exams), orderBy('order'))
   257→   257→      )
   258→   258→
   259→   259→      const examsByCategory: Record<string, FirestoreExam[]> = {}
   260→   260→      examSnap.docs.forEach((d) => {
   261→   261→        const exam = { id: d.id, ...d.data() } as FirestoreExam
   262→   262→        if (!examsByCategory[exam.categoryId]) examsByCategory[exam.categoryId] = []
   263→   263→        examsByCategory[exam.categoryId].push(exam)
   264→   264→      })
   265→   265→
   266→   266→      return catSnap.docs.map((d) => {
   267→   267→        const cat = { id: d.id, ...d.data() } as FirestoreExamCategory
   268→   268→        const catExams = examsByCategory[cat.id] || []
   269→   269→        return {
   270→   270→          id: cat.id,
   271→   271→          name: cat.name,
   272→   272→          slug: cat.slug,
   273→   273→          icon: cat.icon,
   274→   274→          description: cat.description,
   275→   275→          order: cat.order,
   276→   276→          exams: catExams.map((e) => ({
   277→   277→            id: e.id,
   278→   278→            name: e.name,
   279→   279→            slug: e.slug,
   280→   280→            description: e.description,
   281→   281→            totalQuestions: e.totalQuestions,
   282→   282→            duration: e.duration,
   283→   283→            markingScheme: e.markingScheme,
   284→   284→            order: e.order,
   285→   285→            testCount: e.testCount,
   286→   286→          })),
   287→   287→        } as LocalExamCategory
   288→   288→      })
   289→   289→    },
   290→   290→    () => getLocalCategories()
   291→   291→  )
   292→   292→}

   327→   327→export async function deleteCategory(id: string): Promise<void> {
   328→   328→  try {
   329→   329→    // Delete associated exams first
   330→   330→    const examSnap = await getDocs(
   331→   331→      query(collection(db, COLLECTIONS.exams), where('categoryId', '==', id))
   332→   332→    )
   333→   333→    const batch = writeBatch(db)
   334→   334→    examSnap.docs.forEach((d) => batch.delete(d.ref))
   335→   335→    batch.delete(doc(db, COLLECTIONS.categories, id))
   336→   336→    await batch.commit()
   337→   337→  } catch (error) {
   338→   338→    console.error('[Firestore] deleteCategory error:', error)
   339→   339→    throw error
   340→   340→  }

   347→   347→/**
   348→   348→ * Get exams, optionally filtered by category.
   349→   349→ */
   350→   350→export async function getExams(categoryId?: string): Promise<LocalExam[]> {
   351→   351→  return firestoreOperation(
   352→   352→    async () => {
   353→   353→      const constraints: QueryConstraint[] = [orderBy('order')]
   354→   354→      if (categoryId) constraints.push(where('categoryId', '==', categoryId))
   355→   355→
   356→   356→      const snap = await getDocs(query(collection(db, COLLECTIONS.exams), ...constraints))
   357→   357→      return snap.docs.map((d) => {
   358→   358→        const e = { id: d.id, ...d.data() } as FirestoreExam
   359→   359→        return {
   360→   360→          id: e.id,
   361→   361→          name: e.name,
   362→   362→          slug: e.slug,
   363→   363→          description: e.description,
   364→   364→          totalQuestions: e.totalQuestions,
   365→   365→          duration: e.duration,
   366→   366→          markingScheme: e.markingScheme,
   367→   367→          order: e.order,
   368→   368→          testCount: e.testCount,
   369→   369→        } as LocalExam
   370→   370→      })
   371→   371→    },
   372→   372→    () => {
   373→   373→      const cats = getLocalCategories()
   374→   374→      const allExams = cats.flatMap((c) => c.exams)
   375→   375→      if (categoryId) {
   376→   376→        // Filter by category: match category id from parent
   377→   377→        const cat = cats.find(c => c.id === categoryId)
   378→   378→        return cat ? cat.exams : []
   379→   379→      }
   380→   380→      return allExams
   381→   381→    }
   382→   382→  )
   383→   383→}

   415→   415→/**
   416→   416→ * Delete an exam and all its associated tests and questions.
   417→   417→ */
   418→   418→export async function deleteExam(id: string): Promise<void> {
   419→   419→  try {
   420→   420→    // Delete associated tests
   421→   421→    const testSnap = await getDocs(
   422→   422→      query(collection(db, COLLECTIONS.tests), where('examId', '==', id))
   423→   423→    )
   424→   424→    const batch = writeBatch(db)
   425→   425→    for (const testDoc of testSnap.docs) {
   426→   426→      // Delete associated questions
   427→   427→      const qSnap = await getDocs(
   428→   428→        query(collection(db, COLLECTIONS.questions), where('testId', '==', testDoc.id))
   429→   429→      )
   430→   430→      qSnap.docs.forEach((q) => batch.delete(q.ref))
   431→   431→      batch.delete(testDoc.ref)
   432→   432→    }
   433→   433→    batch.delete(doc(db, COLLECTIONS.exams, id))
   434→   434→    await batch.commit()
   435→   435→  } catch (error) {
   436→   436→    console.error('[Firestore] deleteExam error:', error)
   437→   437→    throw error
   438→   438→  }
   439→   439→}

   467→   467→          difficulty: t.difficulty,
   468→   468→          isFree: t.isFree,

   485→   485→export async function getTestById(id: string): Promise<LocalTest | null> {
   486→   486→  return firestoreOperation(
   487→   487→    async () => {
   488→   488→      const testDoc = await getDoc(doc(db, COLLECTIONS.tests, id))
   489→   489→      if (!testDoc.exists()) return null
   490→   490→
   491→   491→      const t = { id: testDoc.id, ...testDoc.data() } as FirestoreTest
   492→   492→
   493→   493→      // Fetch questions for this test
   494→   494→      const qSnap = await getDocs(
   495→   495→        query(collection(db, COLLECTIONS.questions), where('testId', '==', id), orderBy('order'))
   496→   496→      )
   497→   497→      const questions: LocalQuestion[] = qSnap.docs.map((qd) => {
   498→   498→        const q = { id: qd.id, ...qd.data() } as FirestoreQuestion
   499→   499→        return {
   500→   500→          id: q.id,
   501→   501→          questionText: q.questionText,
   502→   502→          questionImage: q.questionImage,
   503→   503→          optionA: q.optionA,
   504→   504→          optionB: q.optionB,
   505→   505→          optionC: q.optionC,
   506→   506→          optionD: q.optionD,
   507→   507→          correctAnswer: q.correctAnswer,
   508→   508→          explanation: q.explanation,
   509→   509→          subject: q.subject,
   510→   510→          order: q.order,
   511→   511→        }
   512→   512→      })
   513→   513→
   514→   514→      return {
   515→   515→        id: t.id,
   516→   516→        title: t.title,
   517→   517→        slug: t.slug,
   518→   518→        description: t.description,
   519→   519→        totalQuestions: t.totalQuestions,
   520→   520→        duration: t.duration,
   521→   521→        markingCorrect: t.markingCorrect,
   522→   522→        markingWrong: t.markingWrong,
   523→   523→        markingSkipped: t.markingSkipped,
   524→   524→        difficulty: t.difficulty,
   525→   525→        isFree: t.isFree,
   526→   526→        isLive: t.isLive,
   527→   527→        exam: { id: t.examId, name: t.examName, slug: t.examSlug },
   528→   528→        questions,
   529→   529→      } as LocalTest
   530→   530→    },
   531→   531→    () => getLocalTestById(id) ?? null
   532→   532→  )
   533→   533→}

   692→   692→        testId,
   693→   693→      }

   713→   713→export async function getResults(
   714→   714→  testId?: string,
   715→   715→  userId?: string
   716→   716→): Promise<TestResult[]> {
   717→   717→  return firestoreOperation(
   718→   718→    async () => {
   719→   719→      const constraints: QueryConstraint[] = [orderBy('createdAt', 'desc')]
   720→   720→      if (testId) constraints.push(where('testId', '==', testId))
   721→   721→      if (userId) constraints.push(where('userId', '==', userId))
   722→   722→
   723→   723→      const snap = await getDocs(query(collection(db, COLLECTIONS.results), ...constraints))
   724→   724→      return snap.docs.map((d) => {
   725→   725→        const r = d.data() as FirestoreTestResult
   726→   726→        return {
   727→   727→          id: d.id,
   728→   728→          testId: r.testId,
   729→   729→          testName: r.testName,
   730→   730→          examName: r.examName,
   731→   731→          userId: r.userId,
   732→   732→          correctCount: r.correctCount,
   733→   733→          wrongCount: r.wrongCount,
   734→   734→          skippedCount: r.skippedCount,
   735→   735→          score: r.score,
   736→   736→          maxScore: r.maxScore,
   737→   737→          timeTaken: r.timeTaken,
   738→   738→          totalQuestions: r.totalQuestions,
   739→   739→          answers: r.answers,
   740→   740→          createdAt:
   741→   741→            r.createdAt instanceof Timestamp
   742→   742→              ? r.createdAt.toDate().toISOString()
   743→   743→              : String(r.createdAt),
   744→   744→        } as TestResult
   745→   745→      })
   746→   746→    },
   747→   747→    () => {
   748→   748→      const allResults = getLocalResults()
   749→   749→      if (testId) return allResults.filter((r: TestResult) => r.testId === testId)
   750→   750→      return allResults
   751→   751→    }
   752→   752→  )
   753→   753→}

   783→   783→export async function getLeaderboard(testId: string): Promise<TestResult[]> {
   784→   784→  return firestoreOperation(
   785→   785→    async () => {

   787→   787→        query(
   788→   788→          collection(db, COLLECTIONS.results),
   789→   789→          where('testId', '==', testId),
   790→   790→          orderBy('score', 'desc'),
   791→   791→          limit(50)
   792→   792→        )
   793→   793→      )
   794→   794→      const results: TestResult[] = snap.docs.map((d) => {
   795→   795→        const r = d.data() as FirestoreTestResult
   796→   796→        return {
   797→   797→          id: d.id,
   798→   798→          testId: r.testId,
   799→   799→          testName: r.testName,
   800→   800→          examName: r.examName,
   801→   801→          userId: r.userId,
   802→   802→          correctCount: r.correctCount,
   803→   803→          wrongCount: r.wrongCount,
   804→   804→          skippedCount: r.skippedCount,
   805→   805→          score: r.score,
   806→   806→          maxScore: r.maxScore,
   807→   807→          timeTaken: r.timeTaken,
   808→   808→          totalQuestions: r.totalQuestions,
   809→   809→          answers: r.answers,
   810→   810→          createdAt:
   811→   811→            r.createdAt instanceof Timestamp
   812→   812→              ? r.createdAt.toDate().toISOString()
   813→   813→              : String(r.createdAt),
   814→   814→        }
   815→   815→      })
   816→   816→      // Secondary sort by timeTaken ascending (Firestore only supports one orderBy)
   817→   817→      return results.sort((a, b) => b.score - a.score || a.timeTaken - b.timeTaken)
   818→   818→    },
   819→   819→    () => getLocalLeaderboard(testId)
   820→   820→  )
   821→   821→}

  1136→  1136→/**
  1137→  1137→ * Get daily stats breakdown for the last N days.
  1138→  1138→ * Groups results by date and computes daily averages.
  1139→  1139→ */
  1140→  1140→export async function getDailyStats(days: number = 7): Promise<DailyStats[]> {
  1141→  1141→  try {
  1142→  1142→    const now = new Date()
  1143→  1143→    const since = new Date(now.getTime() - days * 24 * 60 * 60 * 1000)
  1144→  1144→
  1145→  1145→    const resultsSnap = await getDocs(
  1146→  1146→      query(
  1147→  1147→        collection(db, COLLECTIONS.results),
  1148→  1148→        orderBy('createdAt', 'desc'),
  1149→  1149→        limit(500)
  1150→  1150→      )
  1151→  1151→    )
  1152→  1152→
  1153→  1153→    // Group by date
  1154→  1154→    const dailyMap: Record<
  1155→  1155→      string,
  1156→  1156→      { results: number; totalScore: number; users: Set<string> }
  1157→  1157→    > = {}
  1158→  1158→
  1159→  1159→    // Initialize all days with zero
  1160→  1160→    for (let i = 0; i < days; i++) {
  1161→  1161→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1162→  1162→      const key = d.toISOString().split('T')[0]
  1163→  1163→      dailyMap[key] = { results: 0, totalScore: 0, users: new Set() }
  1164→  1164→    }
  1165→  1165→
  1166→  1166→    resultsSnap.docs.forEach((d) => {
  1167→  1167→      const r = d.data() as FirestoreTestResult
  1168→  1168→      const dateStr =
  1169→  1169→        r.createdAt instanceof Timestamp
  1170→  1170→          ? r.createdAt.toDate().toISOString().split('T')[0]
  1171→  1171→          : String(r.createdAt).split('T')[0]
  1172→  1172→
  1173→  1173→      if (dailyMap[dateStr]) {
  1174→  1174→        dailyMap[dateStr].results++
  1175→  1175→        dailyMap[dateStr].totalScore += r.score
  1176→  1176→        dailyMap[dateStr].users.add(r.userId)
  1177→  1177→      }
  1178→  1178→    })
  1179→  1179→
  1180→  1180→    return Object.entries(dailyMap)
  1181→  1181→      .sort(([a], [b]) => a.localeCompare(b))
  1182→  1182→      .map(([date, stats]) => ({
  1183→  1183→        date,
  1184→  1184→        results: stats.results,
  1185→  1185→        avgScore: stats.results > 0 ? Math.round(stats.totalScore / stats.results) : 0,
  1186→  1186→        uniqueUsers: stats.users.size,
  1187→  1187→      }))
  1188→  1188→  } catch (error) {
  1189→  1189→    console.error('[Firestore] getDailyStats error:', error)
  1190→  1190→    // Return empty daily stats
  1191→  1191→    const stats: DailyStats[] = []
  1192→  1192→    const now = new Date()
  1193→  1193→    for (let i = days - 1; i >= 0; i--) {
  1194→  1194→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1195→  1195→      stats.push({
  1196→  1196→        date: d.toISOString().split('T')[0],
  1197→  1197→        results: 0,
  1198→  1198→        avgScore: 0,
  1199→  1199→        uniqueUsers: 0,
  1200→  1200→      })
  1201→  1201→    }
  1202→  1202→    return stats
  1203→  1203→  }
  1204→  1204→}
