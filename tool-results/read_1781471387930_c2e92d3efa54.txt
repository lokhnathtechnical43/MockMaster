   104→/** Share with Web Share API fallback to clipboard copy */
   105→async function shareApp(title: string, text: string, url: string) {
   106→  if (navigator.share) {
   107→    try {
   108→      await navigator.share({ title, text, url })
   109→    } catch {}
   110→  } else {
   111→    // Desktop fallback: copy link to clipboard
   112→    try {
   113→      await navigator.clipboard.writeText(`${text} ${url}`)
   114→      alert('Link copied to clipboard!')
   115→    } catch {
   116→      // Ultimate fallback
   117→      prompt('Copy this link:', url)
   118→    }
   119→  }
   120→}

   688→  // --- Stats from localStorage ---
   689→  function getUserStats() {
   690→    try {
   691→      // Use getLocalResults() which reads from the correct 'mockmaster_results' key
   692→      const allData: TestResult[] = getLocalResults()
   693→      if (allData.length === 0) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   694→      const userId = auth.getUserId()
   695→      const userResults = userId ? allData.filter((r: TestResult) => r.userId === userId) : allData
   696→      const testsTaken = userResults.length
   697→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.maxScore > 0 ? (r.score / r.maxScore) * 100 : 0), 0) / testsTaken) : 0
   698→      // Calculate best rank from leaderboard data across all tests taken by user
   699→      let bestRank = 0
   700→      if (testsTaken > 0) {
   701→        const uniqueTests = [...new Set(userResults.map(r => r.testId))]
   702→        uniqueTests.forEach(testId => {
   703→          const testResults = allData
   704→            .filter(r => r.testId === testId)
   705→            .sort((a, b) => b.score - a.score || a.timeTaken - b.timeTaken)
   706→          const rank = testResults.findIndex(r => r.userId === userId) + 1
   707→          if (rank > 0 && (bestRank === 0 || rank < bestRank)) bestRank = rank
   708→        })
   709→      }
   710→      return { testsTaken, avgScore, bestRank }
   711→    } catch {
   712→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   713→    }
   714→  }
