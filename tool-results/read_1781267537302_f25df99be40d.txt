   304→  // --- Test Functions ---
   305→  function startTest(test: LocalTest) {
   306→    const fullTest = getTestById(test.id)
   307→    if (!fullTest) return
   308→    setSelectedTest(fullTest)
   309→    setAnswers({})

   316→
   317→  function handleFinishTest() {
   318→    if (timerRef.current) clearInterval(timerRef.current)
   319→    setTestActive(false)
   320→
   321→    if (!selectedTest) return
   322→
   323→    const test = selectedTest
   324→    const totalQuestions = test.questions.length
   325→    let correctCount = 0
   326→    let wrongCount = 0
   327→    let skippedCount = 0
   328→
   329→    test.questions.forEach(q => {
   330→      const userAnswer = answers[q.id]
   331→      if (!userAnswer) {
   332→        skippedCount++
   333→      } else if (userAnswer === q.correctAnswer) {
   334→        correctCount++
   335→      } else {
   336→        wrongCount++
   337→      }
   338→    })
   339→
   340→    const score = correctCount * test.markingCorrect - wrongCount * Math.abs(test.markingWrong)
   341→    const maxScore = totalQuestions * test.markingCorrect
   342→    const timeTaken = test.duration * 60 - timeLeft
   343→
   344→    const result = saveResult({
   345→      testId: test.id,
   346→      testName: test.title,
   347→      examName: test.exam.name,
   348→      userId: auth.getUserId() || 'anonymous',
   349→      correctCount,
   350→      wrongCount,
   351→      skippedCount,
   352→      score,
   353→      maxScore,
   354→      timeTaken,
   355→      totalQuestions,
   356→      answers,
   357→    })
   358→
   359→    setLastResult(result)
   360→    navigateTo('results')
   361→  }

   384→  // --- Stats from localStorage ---
   385→  function getUserStats() {
   386→    try {
   387→      const data = localStorage.getItem('examprep_results')
   388→      if (!data) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   389→      const results: TestResult[] = JSON.parse(data)
   390→      const userId = auth.getUserId()
   391→      const userResults = userId ? results.filter((r: TestResult) => r.userId === userId) : results
   392→      const testsTaken = userResults.length
   393→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.score / r.maxScore) * 100, 0) / testsTaken) : 0
   394→      const bestRank = 1 // simplified
   395→      return { testsTaken, avgScore, bestRank }
   396→    } catch {
   397→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   398→    }
   399→  }
