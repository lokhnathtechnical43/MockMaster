    57→// ===== Helper Functions =====
    58→function formatTime(seconds: number): string {
    59→  const m = Math.floor(seconds / 60)
    60→  const s = seconds % 60
    61→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    62→}

   171→  // --- Test Functions ---
   172→  function startTest(test: LocalTest) {
   173→    const fullTest = getTestById(test.id)
   174→    if (!fullTest) return
   175→    setSelectedTest(fullTest)
   176→    setAnswers({})
   177→    setMarkedForReview(new Set())
   178→    setCurrentQuestionIndex(0)
   179→    setTimeLeft(fullTest.duration * 60)
   180→    setTestActive(true)
   181→    navigateTo('test-taking')

   183→
   184→  function handleFinishTest() {
   185→    if (timerRef.current) clearInterval(timerRef.current)
   186→    setTestActive(false)
   187→
   188→    if (!selectedTest) return
   189→
   190→    const test = selectedTest
   191→    const totalQuestions = test.questions.length
   192→    let correctCount = 0
   193→    let wrongCount = 0
   194→    let skippedCount = 0
   195→
   196→    test.questions.forEach(q => {
   197→      const userAnswer = answers[q.id]
   198→      if (!userAnswer) {
   199→        skippedCount++
   200→      } else if (userAnswer === q.correctAnswer) {
   201→        correctCount++
   202→      } else {
   203→        wrongCount++
   204→      }
   205→    })
   206→
   207→    const score = correctCount * test.markingCorrect - wrongCount * Math.abs(test.markingWrong)
   208→    const maxScore = totalQuestions * test.markingCorrect
   209→    const timeTaken = test.duration * 60 - timeLeft
   210→
   211→    const result = saveResult({
   212→      testId: test.id,
   213→      testName: test.title,
   214→      examName: test.exam.name,
   215→      userId: auth.getUserId() || 'anonymous',
   216→      correctCount,
   217→      wrongCount,
   218→      skippedCount,
   219→      score,
   220→      maxScore,
   221→      timeTaken,
   222→      totalQuestions,
   223→      answers,
   224→    })
   225→
   226→    setLastResult(result)
   227→    navigateTo('results')
   228→  }

   234→  function clearAnswer(questionId: string) {
   235→    setAnswers(prev => {

   239→    })

   242→  function toggleReview(questionId: string) {
   243→    setMarkedForReview(prev => {
   244→      const next = new Set(prev)
   245→      if (next.has(questionId)) next.delete(questionId)
   246→      else next.add(questionId)
   247→      return next
   248→    })
   249→  }

   251→  // --- Stats from localStorage ---
   252→  function getUserStats() {
   253→    try {
   254→      const data = localStorage.getItem('examprep_results')
   255→      if (!data) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   256→      const results: TestResult[] = JSON.parse(data)
   257→      const userId = auth.getUserId()
   258→      const userResults = userId ? results.filter((r: TestResult) => r.userId === userId) : results
   259→      const testsTaken = userResults.length
   260→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.score / r.maxScore) * 100, 0) / testsTaken) : 0
   261→      const bestRank = 1 // simplified
   262→      return { testsTaken, avgScore, bestRank }
   263→    } catch {
   264→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   265→    }
   266→  }
