    20→export interface LocalExam {
    21→  id: string
    22→  name: string
    23→  slug: string
    24→  description: string
    25→  totalQuestions: number
    26→  duration: number
    27→  markingScheme: string

    34→export interface LocalTest {
    35→  id: string
    36→  title: string
    37→  slug: string
    38→  description: string
    39→  totalQuestions: number
    40→  duration: number

    50→  examName?: string
    51→  examSlug?: string
    52→  imageUrl?: string
    53→}
    54→
    55→export interface LocalQuestion {
    56→  id: string
    57→  questionText: string
    58→  questionImage?: string | null
    59→  optionA: string
    60→  optionB: string
    61→  optionC: string
    62→  optionD: string
    63→  correctAnswer: string
    64→  explanation?: string | null
    65→  subject?: string | null
    66→  order?: number
    67→}
    68→
    69→export interface TestResult {
    70→  id: string
    71→  testId: string
    72→  testName: string
    73→  examName?: string
    74→  userId: string
    75→  userName?: string
    76→  score: number
    77→  maxScore: number
    78→  correctCount: number
    79→  wrongCount: number
    80→  skippedCount: number
    81→  timeTaken: number
    82→  totalQuestions?: number
    83→  answers: Record<string, string>
    84→  createdAt: string
    85→  mode?: 'real' | 'practice'
    86→}

    92→function makeQ(id: string, text: string, a: string, b: string, c: string, d: string, correct: string, explanation: string, subject: string): LocalQuestion {
    93→  return { id, questionText: text, optionA: a, optionB: b, optionC: c, optionD: d, correctAnswer: correct, explanation, subject, order: parseInt(id.split('-').pop() || '0') }
    94→}

   418→export function getResults(): TestResult[] {
   419→  if (typeof window === 'undefined') return []
   420→  try {
   421→    const stored = localStorage.getItem('mockmaster_results')
   422→    if (stored) return JSON.parse(stored)
   423→  } catch {}
   424→  return []
   425→}
   426→
   427→export function saveResult(data: Omit<TestResult, 'id' | 'createdAt'>): TestResult {
   428→  const result: TestResult = {
   429→    ...data,
   430→    id: Date.now().toString(),
   431→    createdAt: new Date().toISOString(),
   432→  }
   433→  if (typeof window !== 'undefined') {
   434→    try {
   435→      const existing = getResults()
   436→      localStorage.setItem('mockmaster_results', JSON.stringify([result, ...existing]))
   437→    } catch {}
   438→  }
   439→  return result
   440→}
   441→
   442→export function getLeaderboard(testId: string): TestResult[] {
   443→  return getResults()
   444→    .filter((r) => r.testId === testId)
   445→    .sort((a, b) => b.score - a.score)
   446→    .slice(0, 50)
   447→}
