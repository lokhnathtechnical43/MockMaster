   663→   663→  function getUserStats() {
   664→   664→    try {
   665→   665→      // Use getLocalResults() which reads from the correct 'mockmaster_results' key
   666→   666→      const allData: TestResult[] = getLocalResults()
   667→   667→      if (allData.length === 0) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   668→   668→      const userId = auth.getUserId()
   669→   669→      const userResults = userId ? allData.filter((r: TestResult) => r.userId === userId) : allData
   670→   670→      const testsTaken = userResults.length
   671→   671→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.maxScore > 0 ? (r.score / r.maxScore) * 100 : 0), 0) / testsTaken) : 0
   672→   672→      // Calculate best rank from leaderboard data across all tests taken by user
   673→   673→      let bestRank = 0
   674→   674→      if (testsTaken > 0) {
   675→   675→        const uniqueTests = [...new Set(userResults.map(r => r.testId))]
   676→   676→        uniqueTests.forEach(testId => {
   677→   677→          const testResults = allData
   678→   678→            .filter(r => r.testId === testId)
   679→   679→            .sort((a, b) => b.score - a.score || a.timeTaken - b.timeTaken)
   680→   680→          const rank = testResults.findIndex(r => r.userId === userId) + 1
   681→   681→          if (rank > 0 && (bestRank === 0 || rank < bestRank)) bestRank = rank
   682→   682→        })
   683→   683→      }
   684→   684→      return { testsTaken, avgScore, bestRank }
   685→   685→    } catch {
   686→   686→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   687→   687→    }
   688→   688→  }
