   112→   112→function makeTest(title: string, exam: LocalExam, questions: LocalQuestion[]): LocalTest {
   113→   113→  const testId = uid()
   114→   114→  return {
   115→   115→    id: testId,
   116→   116→    title,
   117→   117→    slug: title.toLowerCase().replace(/[^a-z0-9]+/g, '-'),
   118→   118→    description: `Practice test for ${exam.name}`,
   119→   119→    totalQuestions: questions.length,
   120→   120→    duration: 10,
   121→   121→    markingCorrect: 1.0,
   122→   122→    markingWrong: 0.0,
   123→   123→    markingSkipped: 0.0,
   124→   124→    difficulty: 'mixed',
   125→   125→    isFree: true,
   126→   126→    isLive: true,
   127→   127→    exam: { id: exam.id, name: exam.name, slug: exam.slug },
   128→   128→    questions: questions.map((qq, i) => ({ ...qq, order: i + 1 }))
   129→   129→  }
   130→   130→}

   466→   466→export function saveResult(result: Omit<TestResult, 'id' | 'createdAt'>): TestResult {
   467→   467→  const results = getResults()
   468→   468→  const newResult: TestResult = {
   469→   469→    ...result,
   470→   470→    id: `result_${Date.now()}_${Math.random().toString(36).slice(2, 6)}`,
   471→   471→    createdAt: new Date().toISOString()
   472→   472→  }
   473→   473→  results.push(newResult)
   474→   474→  localStorage.setItem(RESULTS_KEY, JSON.stringify(results))
   475→   475→  return newResult
   476→   476→}
   477→   477→
   478→   478→export function getResults(testId?: string): TestResult[] {
   479→   479→  try {
   480→   480→    const data = localStorage.getItem(RESULTS_KEY)
   481→   481→    if (!data) return []
   482→   482→    const results: TestResult[] = JSON.parse(data)
   483→   483→    if (testId) return results.filter(r => r.testId === testId)
   484→   484→    return results
   485→   485→  } catch {
   486→   486→    return []
   487→   487→  }
   488→   488→}

   490→   490→export function getLeaderboard(testId: string): TestResult[] {
   491→   491→  return getResults(testId).sort((a, b) => b.score - a.score || a.timeTaken - b.timeTaken).slice(0, 50)
   492→   492→}
