   205→// ============================================================
   206→// Helper: Safe Firestore call with fallback
   207→// ============================================================
   208→
   209→async function firestoreOperation<T>(
   210→  firestoreFn: () => Promise<T>,
   211→  fallbackFn: () => T
   212→): Promise<T> {
   213→  if (!useFirestore || !isFirebaseReady() || !db) {
   214→    return fallbackFn()
   215→  }
   216→  try {
   217→    return await firestoreFn()
   218→  } catch (error) {
   219→    console.error('[Firestore] Operation failed, falling back to local data:', error)
   220→    return fallbackFn()
   221→  }
   222→}

   228→/**
   229→ * Get all categories with their nested exams.
   230→ * Firestore: fetches categories & exams separately, then merges.
   231→ * Fallback: returns local-data categories.
   232→ */
   233→export async function getCategories(): Promise<LocalExamCategory[]> {
   234→  return firestoreOperation(
   235→    async () => {
   236→      const catSnap = await getDocs(
   237→        query(collection(db, COLLECTIONS.categories), orderBy('order'))
   238→      )
   239→      const examSnap = await getDocs(
   240→        query(collection(db, COLLECTIONS.exams), orderBy('order'))
   241→      )
   242→
   243→      const examsByCategory: Record<string, FirestoreExam[]> = {}
   244→      examSnap.docs.forEach((d) => {
   245→        const exam = { id: d.id, ...d.data() } as FirestoreExam
   246→        if (!examsByCategory[exam.categoryId]) examsByCategory[exam.categoryId] = []
   247→        examsByCategory[exam.categoryId].push(exam)
   248→      })
   249→
   250→      return catSnap.docs.map((d) => {
   251→        const cat = { id: d.id, ...d.data() } as FirestoreExamCategory
   252→        const catExams = examsByCategory[cat.id] || []
   253→        return {
   254→          id: cat.id,
   255→          name: cat.name,
   256→          slug: cat.slug,
   257→          icon: cat.icon,
   258→          description: cat.description,
   259→          order: cat.order,
   260→          exams: catExams.map((e) => ({
   261→            id: e.id,
   262→            name: e.name,
   263→            slug: e.slug,
   264→            description: e.description,
   265→            totalQuestions: e.totalQuestions,
   266→            duration: e.duration,
   267→            markingScheme: e.markingScheme,
   268→            order: e.order,
   269→            testCount: e.testCount,
   270→          })),
   271→        } as LocalExamCategory
   272→      })
   273→    },
   274→    () => getLocalCategories()
   275→  )
   276→}

   331→/**
   332→ * Get exams, optionally filtered by category.
   333→ */
   334→export async function getExams(categoryId?: string): Promise<LocalExam[]> {
   335→  return firestoreOperation(
   336→    async () => {
   337→      const constraints: QueryConstraint[] = [orderBy('order')]
   338→      if (categoryId) constraints.push(where('categoryId', '==', categoryId))
   339→
   340→      const snap = await getDocs(query(collection(db, COLLECTIONS.exams), ...constraints))
   341→      return snap.docs.map((d) => {
   342→        const e = { id: d.id, ...d.data() } as FirestoreExam
   343→        return {
   344→          id: e.id,
   345→          name: e.name,
   346→          slug: e.slug,
   347→          description: e.description,
   348→          totalQuestions: e.totalQuestions,
   349→          duration: e.duration,
   350→          markingScheme: e.markingScheme,
   351→          order: e.order,
   352→          testCount: e.testCount,
   353→        } as LocalExam
   354→      })
   355→    },
   356→    () => {
   357→      const cats = getLocalCategories()
   358→      const allExams = cats.flatMap((c) => c.exams)
   359→      if (categoryId) return allExams // Local data has exams nested; filter is approximate
   360→      return allExams
   361→    }
   362→  )
   363→}

   395→/**
   396→ * Delete an exam and all its associated tests and questions.
   397→ */
   398→export async function deleteExam(id: string): Promise<void> {
   399→  try {
   400→    // Delete associated tests
   401→    const testSnap = await getDocs(
   402→      query(collection(db, COLLECTIONS.tests), where('examId', '==', id))
   403→    )
   404→    const batch = writeBatch(db)
   405→    for (const testDoc of testSnap.docs) {
   406→      // Delete associated questions
   407→      const qSnap = await getDocs(
   408→        query(collection(db, COLLECTIONS.questions), where('testId', '==', testDoc.id))
   409→      )
   410→      qSnap.docs.forEach((q) => batch.delete(q.ref))
   411→      batch.delete(testDoc.ref)
   412→    }
   413→    batch.delete(doc(db, COLLECTIONS.exams, id))
   414→    await batch.commit()
   415→  } catch (error) {
   416→    console.error('[Firestore] deleteExam error:', error)
   417→    throw error
   418→  }
   419→}

   462→/**
   463→ * Get a single test by ID, including its questions.
   464→ */
   465→export async function getTestById(id: string): Promise<LocalTest | null> {
   466→  return firestoreOperation(
   467→    async () => {
   468→      const testDoc = await getDoc(doc(db, COLLECTIONS.tests, id))
   469→      if (!testDoc.exists()) return null
   470→
   471→      const t = { id: testDoc.id, ...testDoc.data() } as FirestoreTest
   472→
   473→      // Fetch questions for this test
   474→      const qSnap = await getDocs(
   475→        query(collection(db, COLLECTIONS.questions), where('testId', '==', id), orderBy('order'))
   476→      )
   477→      const questions: LocalQuestion[] = qSnap.docs.map((qd) => {
   478→        const q = { id: qd.id, ...qd.data() } as FirestoreQuestion
   479→        return {
   480→          id: q.id,
   481→          questionText: q.questionText,
   482→          questionImage: q.questionImage,
   483→          optionA: q.optionA,
   484→          optionB: q.optionB,
   485→          optionC: q.optionC,
   486→          optionD: q.optionD,
   487→          correctAnswer: q.correctAnswer,
   488→          explanation: q.explanation,
   489→          subject: q.subject,
   490→          order: q.order,
   491→        }
   492→      })
   493→
   494→      return {
   495→        id: t.id,
   496→        title: t.title,
   497→        slug: t.slug,
   498→        description: t.description,
   499→        totalQuestions: t.totalQuestions,
   500→        duration: t.duration,
   501→        markingCorrect: t.markingCorrect,
   502→        markingWrong: t.markingWrong,
   503→        markingSkipped: t.markingSkipped,
   504→        difficulty: t.difficulty,
   505→        isFree: t.isFree,
   506→        isLive: t.isLive,
   507→        exam: { id: t.examId, name: t.examName, slug: t.examSlug },
   508→        questions,
   509→      } as LocalTest
   510→    },
   511→    () => getLocalTestById(id) ?? null
   512→  )
   513→}

   646→/**
   647→ * Add multiple questions at once using a Firestore batch write.
   648→ * Returns the list of created questions with their generated IDs.
   649→ */
   650→export async function addBatchQuestions(
   651→  testId: string,
   652→  questions: Omit<LocalQuestion, 'id'>[]
   653→): Promise<LocalQuestion[]> {
   654→  try {
   655→    const batch = writeBatch(db)
   656→    const created: LocalQuestion[] = []
   657→
   658→    for (const qData of questions) {
   659→      const docRef = doc(collection(db, COLLECTIONS.questions))
   660→      const firestoreQ: FirestoreQuestion = {
   661→        id: docRef.id,
   662→        questionText: qData.questionText,
   663→        questionImage: qData.questionImage,
   664→        optionA: qData.optionA,
   665→        optionB: qData.optionB,
   666→        optionC: qData.optionC,
   667→        optionD: qData.optionD,
   668→        correctAnswer: qData.correctAnswer,
   669→        explanation: qData.explanation,
   670→        subject: qData.subject,
   671→        order: qData.order,
   672→        testId,
   673→      }
   674→      batch.set(docRef, firestoreQ)
   675→      created.push({ ...qData, id: docRef.id })
   676→    }
   677→
   678→    await batch.commit()
   679→    return created
   680→  } catch (error) {
   681→    console.error('[Firestore] addBatchQuestions error:', error)
   682→    throw error
   683→  }
   684→}

   897→      batch.set(docRef, { ...item, id: docRef.id })
   898→    }
   899→    await batch.commit()
   900→    // Also persist locally for offline access
   901→    saveLocalAnnouncements(data)
   902→  } catch (error) {

  1011→/**
  1012→ * Get daily stats breakdown for the last N days.
  1013→ * Groups results by date and computes daily averages.
  1014→ */
  1015→export async function getDailyStats(days: number = 7): Promise<DailyStats[]> {
  1016→  try {
  1017→    const now = new Date()
  1018→    const since = new Date(now.getTime() - days * 24 * 60 * 60 * 1000)
  1019→
  1020→    const resultsSnap = await getDocs(
  1021→      query(
  1022→        collection(db, COLLECTIONS.results),
  1023→        orderBy('createdAt', 'desc'),
  1024→        limit(500)
  1025→      )
  1026→    )
  1027→
  1028→    // Group by date
  1029→    const dailyMap: Record<
  1030→      string,
  1031→      { results: number; totalScore: number; users: Set<string> }
  1032→    > = {}
  1033→
  1034→    // Initialize all days with zero
  1035→    for (let i = 0; i < days; i++) {
  1036→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1037→      const key = d.toISOString().split('T')[0]
  1038→      dailyMap[key] = { results: 0, totalScore: 0, users: new Set() }
  1039→    }
  1040→
  1041→    resultsSnap.docs.forEach((d) => {
  1042→      const r = d.data() as FirestoreTestResult
  1043→      const dateStr =
  1044→        r.createdAt instanceof Timestamp
  1045→          ? r.createdAt.toDate().toISOString().split('T')[0]
  1046→          : String(r.createdAt).split('T')[0]
  1047→
  1048→      if (dailyMap[dateStr]) {
  1049→        dailyMap[dateStr].results++
  1050→        dailyMap[dateStr].totalScore += r.score
  1051→        dailyMap[dateStr].users.add(r.userId)
  1052→      }
  1053→    })
  1054→
  1055→    return Object.entries(dailyMap)
  1056→      .sort(([a], [b]) => a.localeCompare(b))
  1057→      .map(([date, stats]) => ({
  1058→        date,
  1059→        results: stats.results,
  1060→        avgScore: stats.results > 0 ? Math.round(stats.totalScore / stats.results) : 0,
  1061→        uniqueUsers: stats.users.size,
  1062→      }))
  1063→  } catch (error) {
  1064→    console.error('[Firestore] getDailyStats error:', error)
  1065→    // Return empty daily stats
  1066→    const stats: DailyStats[] = []
  1067→    const now = new Date()
  1068→    for (let i = days - 1; i >= 0; i--) {
  1069→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1070→      stats.push({
  1071→        date: d.toISOString().split('T')[0],
  1072→        results: 0,
  1073→        avgScore: 0,
  1074→        uniqueUsers: 0,
  1075→      })
  1076→    }
  1077→    return stats
  1078→  }
  1079→}
