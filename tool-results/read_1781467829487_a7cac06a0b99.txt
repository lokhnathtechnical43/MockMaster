    29→// Reusable image upload helper
    30→function handleImageFileUpload(
    31→  file: File,
    32→  onSuccess: (dataUrl: string) => void
    33→) {
    34→  if (file.size > 2 * 1024 * 1024) {
    35→    alert('Image size must be less than 2MB')
    36→    return
    37→  }
    38→  const reader = new FileReader()
    39→  reader.onloadend = () => {
    40→    onSuccess(reader.result as string)
    41→  }
    42→  reader.readAsDataURL(file)
    43→}

    45→// Reusable Image Upload Section Component
    46→function ImageUploadSection({
    47→  imageUrl,
    48→  onImageChange,
    49→  label = 'Image',
    50→}: {
    51→  imageUrl: string
    52→  onImageChange: (url: string) => void
    53→  label?: string
    54→}) {
    55→  const fileInputRef = useRef<HTMLInputElement>(null)
    56→
    57→  return (
    58→    <div>
    59→      <p className="text-xs text-gray-500 mb-1.5">{label}</p>
    60→      {imageUrl ? (
    61→        <div className="relative rounded-xl overflow-hidden mb-2">
    62→          <img src={imageUrl} alt="Preview" className="w-full h-28 object-cover rounded-xl" />
    63→          <button
    64→            type="button"
    65→            onClick={() => onImageChange('')}
    66→            className="absolute top-2 right-2 w-6 h-6 rounded-full bg-black/50 flex items-center justify-center"
    67→          >
    68→            <X className="w-3 h-3 text-white" />
    69→          </button>
    70→        </div>
    71→      ) : (
    72→        <div className="flex gap-2">
    73→          <button
    74→            type="button"
    75→            onClick={() => fileInputRef.current?.click()}
    76→            className="flex-1 flex items-center justify-center gap-2 py-2.5 rounded-xl border-2 border-dashed border-gray-200 hover:border-orange-300 hover:bg-orange-50/50 transition-colors text-xs text-gray-500"
    77→          >
    78→            <ImagePlus className="w-4 h-4" /> Upload
    79→          </button>
    80→          <input
    81→            ref={fileInputRef}
    82→            type="file"
    83→            accept="image/*"
    84→            className="hidden"
    85→            onChange={(e) => {
    86→              const file = e.target.files?.[0]
    87→              if (file) handleImageFileUpload(file, onImageChange)
    88→            }}
    89→          />
    90→          <input
    91→            type="text"
    92→            placeholder="Or paste URL..."
    93→            value={imageUrl}
    94→            onChange={(e) => onImageChange(e.target.value)}
    95→            className="flex-1 px-3 py-2 rounded-xl border border-gray-200 text-xs focus:outline-none focus:ring-2 focus:ring-orange-300"
    96→          />
    97→        </div>
    98→      )}
    99→    </div>
   100→  )
   101→}

   121→  const [selectedTestName, setSelectedTestName] = useState('')
   122→
   123→  // Questions

   125→
   126→  // Edit states

   129→  const [editDesc, setEditDesc] = useState('')

   131→

   134→  const [addName, setAddName] = useState('')
   135→  const [addDesc, setAddDesc] = useState('')
   136→  const [addSlug, setAddSlug] = useState('')

   138→
   139→  // Add question

   141→  const [qText, setQText] = useState('')
   142→  const [qA, setQA] = useState('')
   143→  const [qB, setQB] = useState('')
   144→  const [qC, setQC] = useState('')
   145→  const [qD, setQD] = useState('')
   146→  const [qAns, setQAns] = useState('A')
   147→  const [qExplanation, setQExplanation] = useState('')
   148→  const [qSubject, setQSubject] = useState('')
   149→  const [qImageUrl, setQImageUrl] = useState('')
   150→
   151→  // Batch questions
   152→  const [showBatch, setShowBatch] = useState(false)
   153→  const [batchText, setBatchText] = useState('')
   154→
   155→  const loadCategories = useCallback(async () => {

   166→    }
   167→  }, [])
   168→
   169→  const loadExams = useCallback(async (categoryId: string) => {
   170→    setLoading(true)

   173→      setExams(data)
   174→    } catch { setExams([]) } finally { setLoading(false) }

   176→
   177→  const loadTests = useCallback(async (examId: string) => {
   178→    setLoading(true)
   179→    try {

   181→      setTests(data)
   182→    } catch { setTests([]) } finally { setLoading(false) }
   183→  }, [])
   184→

   186→    setLoading(true)
   187→    try {
   188→      const data = await getQuestions(testId)

   195→  const handleSelectCategory = (cat: LocalExamCategory) => {

   201→
   202→  const handleSelectExam = (exam: LocalExam) => {
   203→    setSelectedExamId(exam.id)
   204→    setSelectedExamName(exam.name)
   205→    loadTests(exam.id)
   206→    setViewLevel('tests')

   208→
   209→  const handleSelectTest = (test: LocalTest) => {
   210→    setSelectedTestId(test.id)
   211→    setSelectedTestName(test.title)
   212→    loadQuestions(test.id)
   213→    setViewLevel('questions')
   214→  }
   215→
   216→  const handleBack = () => {
   217→    if (viewLevel === 'questions') { setViewLevel('tests'); setQuestions([]) }

   221→    setEditingId(null)

   223→

   228→      await addCategory({
   229→        name: addName, slug: addSlug || addName.toLowerCase().replace(/\s+/g, '-'),

   232→      })

   254→    if (!addName.trim() || !selectedExamId) return

   256→      await addTest({
   257→        title: addName, slug: addSlug || addName.toLowerCase().replace(/\s+/g, '-'),

   259→        markingCorrect: 1, markingWrong: -0.25, markingSkipped: 0,

   296→          optionB: parts[2]?.trim() || '',
   297→          optionC: parts[3]?.trim() || '',
   298→          optionD: parts[4]?.trim() || '',
   299→          correctAnswer: parts[5]?.trim() || 'A',
   300→          explanation: parts[6]?.trim() || null,
   301→          subject: parts[7]?.trim() || null,
   302→          order: i + 1,
   303→          testId: selectedTestId
   304→        }
   305→      }).filter(q => q.questionText)
   306→      await addBatchQuestions(selectedTestId, qs)
   307→      setBatchText(''); setShowBatch(false)
   308→      loadQuestions(selectedTestId)
   309→    } catch { alert('Failed to add batch questions') }
   310→  }
   311→
   312→  const handleDelete = async (type: string, id: string) => {
   313→    if (!confirm(`Delete this ${type}?`)) return
   314→    try {
   315→      if (type === 'category') { await deleteCategory(id); loadCategories() }
   316→      else if (type === 'exam' && selectedCategoryId) { await deleteExam(id); loadExams(selectedCategoryId) }
   317→      else if (type === 'test' && selectedExamId) { await deleteTest(id); loadTests(selectedExamId) }
   318→      else if (type === 'question' && selectedTestId) { await deleteQuestion(id); loadQuestions(selectedTestId) }
   319→    } catch { alert(`Failed to delete ${type}`) }
   320→  }
   321→
   322→  const handleSeed = async () => {
   323→    if (!confirm('Seed Firestore from local data? This may take a moment.')) return
   324→    setSeeding(true)
   325→    try {
   326→      await seedFirestoreIfEmpty()
   327→      alert('Firestore seeded successfully!')
   328→      loadCategories()
   329→    } catch { alert('Failed to seed Firestore') }
   330→    finally { setSeeding(false) }
   331→  }
   332→
   333→  const handleStartEdit = (id: string, name: string, desc: string, imageUrl?: string) => {
   334→    setEditingId(id)
   335→    setEditName(name)
   336→    setEditDesc(desc || '')
   337→    setEditImageUrl(imageUrl || '')

   339→
   340→  const handleSaveEdit = async (type: string, id: string) => {
   341→    try {
   342→      if (type === 'category') {
   343→        await updateCategory(id, { name: editName, description: editDesc, imageUrl: editImageUrl || undefined })
   344→        loadCategories()
   345→      } else if (type === 'exam') {
   346→        await updateExam(id, { name: editName, description: editDesc, imageUrl: editImageUrl || undefined })
   347→        if (selectedCategoryId) loadExams(selectedCategoryId)
   348→      } else if (type === 'test') {
   349→        await updateTest(id, { title: editName, description: editDesc, imageUrl: editImageUrl || undefined })
   350→        if (selectedExamId) loadTests(selectedExamId)
   351→      }
   352→      setEditingId(null)
   353→    } catch { alert('Failed to update') }
   354→  }
   355→
   356→  return (
   357→    <div className="space-y-4">
   358→      {/* Breadcrumb + Actions */}
   359→      <div className="flex items-center justify-between">
   360→        <div className="flex items-center gap-2 text-sm">
   361→          {viewLevel !== 'categories' && (
   362→            <button onClick={handleBack} className="text-violet-600 hover:text-violet-800 font-medium flex items-center gap-1">
   363→              <ArrowLeft className="w-4 h-4" /> Back
   364→            </button>
   365→          )}
   366→          <span className="text-gray-400">
   367→            {viewLevel === 'categories' ? 'Categories' :
   368→             viewLevel === 'exams' ? `› ${selectedCategoryName} › Exams` :
   369→             viewLevel === 'tests' ? `› ${selectedCategoryName} › ${selectedExamName} › Tests` :
   370→             `› ${selectedCategoryName} › ${selectedExamName} › ${selectedTestName} › Questions`}

   373→        <div className="flex items-center gap-2">

   376→            size="sm"

   378→            onClick={handleSeed}

   395→      {!loading && viewLevel === 'categories' && (
   396→        <div className="space-y-4">
   397→          <div className="flex items-center justify-between">
   398→            <h3 className="font-bold text-sm">Categories ({categories.length})</h3>
   399→            <Button size="sm" className="rounded-xl text-xs" onClick={() => { setShowAdd(!showAdd); setAddName(''); setAddDesc(''); setAddSlug(''); setAddImageUrl('') }}>
   400→              <Plus className="w-3 h-3 mr-1" /> Add Category
   401→            </Button>
   402→          </div>
   403→
   404→          {showAdd && (
   405→            <Card className="border-0 shadow-md bg-orange-50/50">

   409→                <Input placeholder="Description" value={addDesc} onChange={e => setAddDesc(e.target.value)} className="rounded-xl" />

   416→            </Card>

   420→            <Card className="border-0 shadow-sm">
   421→              <CardContent className="p-8 text-center">

   423→                <p className="text-gray-400 text-sm">No categories found</p>

   425→            </Card>
   426→          ) : (
   427→            categories.map(cat => (

   429→                <CardContent className="p-3 flex items-center gap-3">
   430→                  {editingId === cat.id ? (

   432→                      <Input value={editName} onChange={e => setEditName(e.target.value)} className="rounded-xl text-sm" />
   433→                      <Input value={editDesc} onChange={e => setEditDesc(e.target.value)} placeholder="Description" className="rounded-xl text-sm" />
   434→                      <ImageUploadSection imageUrl={editImageUrl} onImageChange={setEditImageUrl} label="Image" />
   435→                      <div className="flex gap-2">
   436→                        <Button size="sm" className="rounded-xl" onClick={() => handleSaveEdit('category', cat.id)}>
   437→                          <Save className="w-3 h-3 mr-1" /> Save
   438→                        </Button>
   439→                        <Button size="sm" variant="outline" className="rounded-xl" onClick={() => setEditingId(null)}>Cancel</Button>
   440→                      </div>
   441→                    </div>
   442→                  ) : (
   443→                    <>
   444→                      <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-orange-100 to-orange-200 flex items-center justify-center flex-shrink-0 overflow-hidden">
   445→                        {cat.imageUrl ? (
   446→                          <img src={cat.imageUrl} alt={cat.name} className="w-full h-full object-cover" />
   447→                        ) : (
   448→                          <span className="text-lg">{cat.icon || '📋'}</span>
   449→                        )}
   450→                      </div>
   451→                      <button className="flex-1 min-w-0 text-left" onClick={() => handleSelectCategory(cat)}>
   452→                        <p className="font-semibold text-sm truncate">{cat.name}</p>
   453→                        <p className="text-gray-400 text-xs truncate">{cat.description || cat.slug}</p>
   454→                        {cat.imageUrl && <Badge className="text-[8px] mt-0.5 bg-blue-100 text-blue-700">Image</Badge>}

   458→                          <Edit3 className="w-3.5 h-3.5 text-blue-500" />
   459→                        </button>

   465→                        </button>
   466→                      </div>

   470→              </Card>
   471→            ))
   472→          )}
   473→        </div>
   474→      )}
   475→
   476→      {/* ===== EXAMS VIEW ===== */}
   477→      {!loading && viewLevel === 'exams' && (
   478→        <div className="space-y-4">
   479→          <div className="flex items-center justify-between">
   480→            <h3 className="font-bold text-sm">Exams in {selectedCategoryName} ({exams.length})</h3>
   481→            <Button size="sm" className="rounded-xl text-xs" onClick={() => { setShowAdd(!showAdd); setAddName(''); setAddDesc(''); setAddSlug(''); setAddImageUrl('') }}>
   482→              <Plus className="w-3 h-3 mr-1" /> Add Exam

   484→          </div>
   485→
   486→          {showAdd && (
   487→            <Card className="border-0 shadow-md bg-orange-50/50">
   488→              <CardContent className="p-4 space-y-2">
   489→                <Input placeholder="Exam name" value={addName} onChange={e => setAddName(e.target.value)} className="rounded-xl" />
   490→                <Input placeholder="Slug" value={addSlug} onChange={e => setAddSlug(e.target.value)} className="rounded-xl" />
   491→                <Input placeholder="Description" value={addDesc} onChange={e => setAddDesc(e.target.value)} className="rounded-xl" />
   492→                <ImageUploadSection imageUrl={addImageUrl} onImageChange={setAddImageUrl} label="Exam Image (optional)" />
