   370→   370→  async function handleFinishTest() {
   371→   371→    if (timerRef.current) clearInterval(timerRef.current)
   372→   372→    setTestActive(false)
   373→   373→
   374→   374→    if (!selectedTest) return
   375→   375→
   376→   376→    const test = selectedTest
   377→   377→    const totalQuestions = test.questions.length
   378→   378→    let correctCount = 0
   379→   379→    let wrongCount = 0
   380→   380→    let skippedCount = 0
   381→   381→
   382→   382→    test.questions.forEach(q => {
   383→   383→      const userAnswer = answers[q.id]
   384→   384→      if (!userAnswer) {
   385→   385→        skippedCount++
   386→   386→      } else if (userAnswer === q.correctAnswer) {
   387→   387→        correctCount++
   388→   388→      } else {
   389→   389→        wrongCount++
   390→   390→      }
   391→   391→    })
   392→   392→
   393→   393→    const score = correctCount * test.markingCorrect - wrongCount * Math.abs(test.markingWrong)
   394→   394→    const maxScore = totalQuestions * test.markingCorrect
   395→   395→    const timeTaken = test.duration * 60 - timeLeft
   396→   396→
   397→   397→    const result = await storeResult({
   398→   398→      testId: test.id,
   399→   399→      testName: test.title,
   400→   400→      examName: test.exam.name,
   401→   401→      userId: auth.getUserId() || 'anonymous',
   402→   402→      correctCount,
   403→   403→      wrongCount,
   404→   404→      skippedCount,
   405→   405→      score,
   406→   406→      maxScore,
   407→   407→      timeTaken,
   408→   408→      totalQuestions,
   409→   409→      answers,
   410→   410→    })
   411→   411→
   412→   412→    setLastResult(result)
   413→   413→    navigateTo('results')
   414→   414→  }

   437→   437→  // --- Stats from localStorage ---
   438→   438→  function getUserStats() {
   439→   439→    try {
   440→   440→      const data = localStorage.getItem('examprep_results')
   441→   441→      if (!data) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   442→   442→      const results: TestResult[] = JSON.parse(data)
   443→   443→      const userId = auth.getUserId()
   444→   444→      const userResults = userId ? results.filter((r: TestResult) => r.userId === userId) : results
   445→   445→      const testsTaken = userResults.length
   446→   446→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.score / r.maxScore) * 100, 0) / testsTaken) : 0
   447→   447→      const bestRank = 1 // simplified
   448→   448→      return { testsTaken, avgScore, bestRank }
   449→   449→    } catch {
   450→   450→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   451→   451→    }
