   357→  // --- Test Functions ---
   358→  async function startTest(test: LocalTest) {
   359→    const fullTest = await fetchTestById(test.id)
   360→    if (!fullTest) return
   361→    setSelectedTest(fullTest)
   362→    setAnswers({})
   363→    setMarkedForReview(new Set())

   365→    setTimeLeft(fullTest.duration * 60)

   367→    navigateTo('test-taking')
   368→  }
   369→
   370→  async function handleFinishTest() {
   371→    if (timerRef.current) clearInterval(timerRef.current)
   372→    setTestActive(false)
   373→
   374→    if (!selectedTest) return
   375→
   376→    const test = selectedTest
   377→    const totalQuestions = test.questions.length
   378→    let correctCount = 0
   379→    let wrongCount = 0
   380→    let skippedCount = 0
   381→
   382→    test.questions.forEach(q => {
   383→      const userAnswer = answers[q.id]
   384→      if (!userAnswer) {
   385→        skippedCount++
   386→      } else if (userAnswer === q.correctAnswer) {
   387→        correctCount++
   388→      } else {
   389→        wrongCount++
   390→      }
   391→    })
   392→
   393→    const score = correctCount * test.markingCorrect - wrongCount * Math.abs(test.markingWrong)
   394→    const maxScore = totalQuestions * test.markingCorrect
   395→    const timeTaken = test.duration * 60 - timeLeft
   396→
   397→    const result = await storeResult({
   398→      testId: test.id,
   399→      testName: test.title,
   400→      examName: test.exam.name,
   401→      userId: auth.getUserId() || 'anonymous',
   402→      correctCount,
   403→      wrongCount,
   404→      skippedCount,
   405→      score,
   406→      maxScore,
   407→      timeTaken,
   408→      totalQuestions,
   409→      answers,
   410→    })
   411→
   412→    setLastResult(result)
   413→    navigateTo('results')
   414→  }

   437→  // --- Stats from localStorage ---
   438→  function getUserStats() {
   439→    try {
   440→      const data = localStorage.getItem('examprep_results')
   441→      if (!data) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   442→      const results: TestResult[] = JSON.parse(data)
   443→      const userId = auth.getUserId()
   444→      const userResults = userId ? results.filter((r: TestResult) => r.userId === userId) : results
   445→      const testsTaken = userResults.length
   446→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.score / r.maxScore) * 100, 0) / testsTaken) : 0
   447→      const bestRank = 1 // simplified
   448→      return { testsTaken, avgScore, bestRank }
   449→    } catch {
   450→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   451→    }
   452→  }
