    26→// Reusable image upload helper
    27→function handleImageFileUpload(
    28→  file: File,
    29→  onSuccess: (dataUrl: string) => void
    30→) {
    31→  if (file.size > 2 * 1024 * 1024) {
    32→    alert('Image size must be less than 2MB')
    33→    return
    34→  }
    35→  const reader = new FileReader()
    36→  reader.onloadend = () => {
    37→    onSuccess(reader.result as string)
    38→  }
    39→  reader.readAsDataURL(file)
    40→}

    42→// Reusable Image Upload Section Component
    43→function ImageUploadSection({
    44→  imageUrl,
    45→  onImageChange,
    46→  label = 'Image',
    47→}: {
    48→  imageUrl: string
    49→  onImageChange: (url: string) => void
    50→  label?: string
    51→}) {
    52→  const fileInputRef = useRef<HTMLInputElement>(null)
    53→
    54→  return (
    55→    <div>
    56→      <p className="text-xs text-gray-500 mb-1.5">{label}</p>
    57→      {imageUrl ? (
    58→        <div className="relative rounded-xl overflow-hidden mb-2">
    59→          <img src={imageUrl} alt="Preview" className="w-full h-28 object-cover rounded-xl" />
    60→          <button
    61→            type="button"
    62→            onClick={() => onImageChange('')}
    63→            className="absolute top-2 right-2 w-6 h-6 rounded-full bg-black/50 flex items-center justify-center"
    64→          >
    65→            <X className="w-3 h-3 text-white" />
    66→          </button>
    67→        </div>
    68→      ) : (
    69→        <div className="flex gap-2">
    70→          <button
    71→            type="button"
    72→            onClick={() => fileInputRef.current?.click()}
    73→            className="flex-1 flex items-center justify-center gap-2 py-2.5 rounded-xl border-2 border-dashed border-gray-200 hover:border-orange-300 hover:bg-orange-50/50 transition-colors text-xs text-gray-500"
    74→          >
    75→            <ImagePlus className="w-4 h-4" /> Upload
    76→          </button>
    77→          <input
    78→            ref={fileInputRef}
    79→            type="file"
    80→            accept="image/*"
    81→            className="hidden"
    82→            onChange={(e) => {
    83→              const file = e.target.files?.[0]
    84→              if (file) handleImageFileUpload(file, onImageChange)
    85→            }}
    86→          />
    87→          <input
    88→            type="text"
    89→            placeholder="Or paste URL..."
    90→            value={imageUrl}
    91→            onChange={(e) => onImageChange(e.target.value)}
    92→            className="flex-1 px-3 py-2 rounded-xl border border-gray-200 text-xs focus:outline-none focus:ring-2 focus:ring-orange-300"
    93→          />
    94→        </div>
    95→      )}
    96→    </div>
    97→  )
    98→}

   100→export default function ExamsTab() {
   101→  const [viewLevel, setViewLevel] = useState<ViewLevel>('categories')
   102→  const [loading, setLoading] = useState(true)
   103→  const [seeding, setSeeding] = useState(false)

   108→  const [selectedCategoryName, setSelectedCategoryName] = useState('')

   112→  const [selectedExamId, setSelectedExamId] = useState<string | null>(null)
   113→  const [selectedExamName, setSelectedExamName] = useState('')

   117→  const [selectedTestId, setSelectedTestId] = useState<string | null>(null)
   118→  const [selectedTestName, setSelectedTestName] = useState('')
   119→
   120→  // Questions
   121→  const [questions, setQuestions] = useState<FirestoreQuestion[]>([])
   122→
   123→  // Edit states
   124→  const [editingId, setEditingId] = useState<string | null>(null)
   125→  const [editName, setEditName] = useState('')
   126→  const [editDesc, setEditDesc] = useState('')
   127→  const [editImageUrl, setEditImageUrl] = useState('')
   128→
   129→  // Add states
   130→  const [showAdd, setShowAdd] = useState(false)
   131→  const [addName, setAddName] = useState('')
   132→  const [addDesc, setAddDesc] = useState('')
   133→  const [addSlug, setAddSlug] = useState('')
   134→  const [addImageUrl, setAddImageUrl] = useState('')
   135→
   136→  // Add question
   137→  const [showAddQuestion, setShowAddQuestion] = useState(false)
   138→  const [qText, setQText] = useState('')
   139→  const [qA, setQA] = useState('')
   140→  const [qB, setQB] = useState('')
   141→  const [qC, setQC] = useState('')
   142→  const [qD, setQD] = useState('')
   143→  const [qAns, setQAns] = useState('A')
   144→  const [qExplanation, setQExplanation] = useState('')
   145→  const [qSubject, setQSubject] = useState('')
   146→  const [qImageUrl, setQImageUrl] = useState('')
   147→
   148→  // Batch questions
   149→  const [showBatch, setShowBatch] = useState(false)
   150→  const [batchText, setBatchText] = useState('')
   151→
   152→  const loadCategories = useCallback(async () => {
   153→    setLoading(true)
   154→    try {
   155→      const cats = await getCategories()
   156→      setCategories(cats.map(c => ({
   157→        id: c.id, name: c.name, slug: c.slug, icon: c.icon,
   158→        description: c.description, order: c.order, imageUrl: c.imageUrl
   159→      })))
   160→    } catch {
   161→      // Fallback to local
   162→      const local = getLocalCategories()
   163→      setCategories(local.map(c => ({
   164→        id: c.id, name: c.name, slug: c.slug, icon: c.icon,
   165→        description: c.description, order: c.order
   166→      })))
   167→    } finally {
   168→      setLoading(false)
   169→    }
   170→  }, [])
   171→
   172→  const loadExams = useCallback(async (categoryId: string) => {
   173→    setLoading(true)
   174→    try {
   175→      const data = await getExams(categoryId)
   176→      setExams(data)
   177→    } catch { setExams([]) } finally { setLoading(false) }
   178→  }, [])
   179→
   180→  const loadTests = useCallback(async (examId: string) => {
   181→    setLoading(true)
   182→    try {
   183→      const data = await getTests(examId)
   184→      setTests(data)
   185→    } catch { setTests([]) } finally { setLoading(false) }
   186→  }, [])
   187→
   188→  const loadQuestions = useCallback(async (testId: string) => {
   189→    setLoading(true)
   190→    try {
   191→      const data = await getQuestions(testId)
   192→      setQuestions(data)
   193→    } catch { setQuestions([]) } finally { setLoading(false) }
   194→  }, [])
   195→
   196→  useEffect(() => { loadCategories() }, [loadCategories])
   197→
   198→  const handleSelectCategory = (cat: FirestoreExamCategory) => {
   199→    setSelectedCategoryId(cat.id)
   200→    setSelectedCategoryName(cat.name)
   201→    loadExams(cat.id)
   202→    setViewLevel('exams')
   203→  }
   204→
   205→  const handleSelectExam = (exam: FirestoreExam) => {
   206→    setSelectedExamId(exam.id)
   207→    setSelectedExamName(exam.name)
   208→    loadTests(exam.id)
   209→    setViewLevel('tests')
   210→  }
   211→
   212→  const handleSelectTest = (test: FirestoreTest) => {
   213→    setSelectedTestId(test.id)
   214→    setSelectedTestName(test.title)
   215→    loadQuestions(test.id)
   216→    setViewLevel('questions')
   217→  }
   218→
   219→  const handleBack = () => {
   220→    if (viewLevel === 'questions') { setViewLevel('tests'); setQuestions([]) }
   221→    else if (viewLevel === 'tests') { setViewLevel('exams'); setTests([]) }
   222→    else if (viewLevel === 'exams') { setViewLevel('categories'); setExams([]) }
   223→    setShowAdd(false)
   224→    setEditingId(null)
   225→  }
   226→
   227→  // --- CRUD Handlers ---

   230→    try {
   231→      await addCategory({
   232→        name: addName, slug: addSlug || addName.toLowerCase().replace(/\s+/g, '-'),
   233→        icon: '📋', description: addDesc, order: categories.length + 1,
   234→        ...(addImageUrl ? { imageUrl: addImageUrl } : {})
   235→      })
   236→      setAddName(''); setAddDesc(''); setAddSlug(''); setAddImageUrl(''); setShowAdd(false)
   237→      loadCategories()
   238→    } catch { alert('Failed to add category') }
   239→  }
   240→
   241→  const handleAddExam = async () => {
   242→    if (!addName.trim() || !selectedCategoryId) return
   243→    try {
   244→      await addExam({
   245→        name: addName, slug: addSlug || addName.toLowerCase().replace(/\s+/g, '-'),
   246→        description: addDesc, totalQuestions: 0, duration: 60,
   247→        markingScheme: '1/-0.25', order: exams.length + 1,
   248→        testCount: 0, categoryId: selectedCategoryId,
   249→        ...(addImageUrl ? { imageUrl: addImageUrl } : {})
   250→      })
   251→      setAddName(''); setAddDesc(''); setAddSlug(''); setAddImageUrl(''); setShowAdd(false)
   252→      loadExams(selectedCategoryId)
   253→    } catch { alert('Failed to add exam') }
   254→  }
   255→
   256→  const handleAddTest = async () => {
   257→    if (!addName.trim() || !selectedExamId) return
   258→    try {
   259→      await addTest({
   260→        title: addName, slug: addSlug || addName.toLowerCase().replace(/\s+/g, '-'),
   261→        description: addDesc, totalQuestions: 0, duration: 30,
   262→        markingCorrect: 1, markingWrong: -0.25, markingSkipped: 0,
   263→        difficulty: 'medium', isFree: true, isLive: true,
   264→        examId: selectedExamId, examName: selectedExamName,
   265→        examSlug: selectedExamName.toLowerCase().replace(/\s+/g, '-'),
   266→        ...(addImageUrl ? { imageUrl: addImageUrl } : {})
   267→      })
   268→      setAddName(''); setAddDesc(''); setAddSlug(''); setAddImageUrl(''); setShowAdd(false)
   269→      loadTests(selectedExamId)
   270→    } catch { alert('Failed to add test') }
   271→  }
   272→
   273→  const handleAddQuestion = async () => {
   274→    if (!qText.trim() || !selectedTestId) return
   275→    try {
   276→      await addQuestion({
   277→        questionText: qText, questionImage: qImageUrl || null,
   278→        optionA: qA, optionB: qB, optionC: qC, optionD: qD,
   279→        correctAnswer: qAns, explanation: qExplanation || null,
   280→        subject: qSubject || null, order: questions.length + 1,
   281→        testId: selectedTestId
   282→      })
   283→      setQText(''); setQA(''); setQB(''); setQC(''); setQD('')
   284→      setQAns('A'); setQExplanation(''); setQSubject(''); setQImageUrl('')
   285→      setShowAddQuestion(false)
   286→      loadQuestions(selectedTestId)
   287→    } catch { alert('Failed to add question') }
   288→  }
   289→
   290→  const handleBatchQuestions = async () => {
   291→    if (!batchText.trim() || !selectedTestId) return
   292→    try {
   293→      const lines = batchText.trim().split('\n').filter(l => l.trim())
   294→      const qs = lines.map((line, i) => {
   295→        const parts = line.split('|')
   296→        return {
   297→          questionText: parts[0]?.trim() || '',
   298→          optionA: parts[1]?.trim() || '',
   299→          optionB: parts[2]?.trim() || '',
   300→          optionC: parts[3]?.trim() || '',
   301→          optionD: parts[4]?.trim() || '',
   302→          correctAnswer: parts[5]?.trim() || 'A',
   303→          explanation: parts[6]?.trim() || null,
   304→          subject: parts[7]?.trim() || null,
   305→          order: i + 1,
   306→          testId: selectedTestId
   307→        }
   308→      }).filter(q => q.questionText)
   309→      await addBatchQuestions(selectedTestId, qs)
   310→      setBatchText(''); setShowBatch(false)
   311→      loadQuestions(selectedTestId)
   312→    } catch { alert('Failed to add batch questions') }
   313→  }
   314→
   315→  const handleDelete = async (type: string, id: string) => {
   316→    if (!confirm(`Delete this ${type}?`)) return
   317→    try {
   318→      if (type === 'category') { await deleteCategory(id); loadCategories() }
   319→      else if (type === 'exam' && selectedCategoryId) { await deleteExam(id); loadExams(selectedCategoryId) }
   320→      else if (type === 'test' && selectedExamId) { await deleteTest(id); loadTests(selectedExamId) }
   321→      else if (type === 'question' && selectedTestId) { await deleteQuestion(id); loadQuestions(selectedTestId) }
   322→    } catch { alert(`Failed to delete ${type}`) }
   323→  }
   324→
   325→  const handleSeed = async () => {
   326→    if (!confirm('Seed Firestore from local data? This may take a moment.')) return
   327→    setSeeding(true)
   328→    try {
   329→      await seedFirestoreIfEmpty()
   330→      alert('Firestore seeded successfully!')
   331→      loadCategories()
   332→    } catch { alert('Failed to seed Firestore') }
   333→    finally { setSeeding(false) }
   334→  }
   335→
   336→  const handleStartEdit = (id: string, name: string, desc: string, imageUrl?: string) => {
   337→    setEditingId(id)
   338→    setEditName(name)
   339→    setEditDesc(desc || '')
   340→    setEditImageUrl(imageUrl || '')
   341→  }
   342→
   343→  const handleSaveEdit = async (type: string, id: string) => {
   344→    try {
   345→      if (type === 'category') {
   346→        await updateCategory(id, { name: editName, description: editDesc, imageUrl: editImageUrl || undefined })
   347→        loadCategories()
   348→      } else if (type === 'exam') {
   349→        await updateExam(id, { name: editName, description: editDesc, imageUrl: editImageUrl || undefined })
   350→        if (selectedCategoryId) loadExams(selectedCategoryId)
   351→      } else if (type === 'test') {
   352→        await updateTest(id, { title: editName, description: editDesc, imageUrl: editImageUrl || undefined })
   353→        if (selectedExamId) loadTests(selectedExamId)
   354→      }
   355→      setEditingId(null)
   356→    } catch { alert('Failed to update') }
   357→  }
   358→
   359→  return (
   360→    <div className="space-y-4">
   361→      {/* Breadcrumb + Actions */}
   362→      <div className="flex items-center justify-between">
   363→        <div className="flex items-center gap-2 text-sm">
   364→          {viewLevel !== 'categories' && (
   365→            <button onClick={handleBack} className="text-violet-600 hover:text-violet-800 font-medium flex items-center gap-1">
   366→              <ArrowLeft className="w-4 h-4" /> Back
   367→            </button>
   368→          )}
   369→          <span className="text-gray-400">
   370→            {viewLevel === 'categories' ? 'Categories' :
   371→             viewLevel === 'exams' ? `› ${selectedCategoryName} › Exams` :

   374→          </span>
   375→        </div>
   376→        <div className="flex items-center gap-2">
   377→          <Button
   378→            variant="outline"
   379→            size="sm"
   380→            className="rounded-xl text-xs"
   381→            onClick={handleSeed}
   382→            disabled={seeding}
   383→          >
   384→            <Database className="w-3 h-3 mr-1" />
   385→            {seeding ? 'Seeding...' : 'Seed Firestore'}
   386→          </Button>
   387→        </div>
   388→      </div>
   389→
   390→      {/* Loading */}
   391→      {loading && (
   392→        <div className="flex items-center justify-center py-12">
   393→          <RefreshCw className="w-6 h-6 text-gray-400 animate-spin" />
   394→        </div>
   395→      )}
   396→
   397→      {/* ===== CATEGORIES VIEW ===== */}
   398→      {!loading && viewLevel === 'categories' && (
   399→        <div className="space-y-4">
   400→          <div className="flex items-center justify-between">
   401→            <h3 className="font-bold text-sm">Categories ({categories.length})</h3>
   402→            <Button size="sm" className="rounded-xl text-xs" onClick={() => { setShowAdd(!showAdd); setAddName(''); setAddDesc(''); setAddSlug(''); setAddImageUrl('') }}>
   403→              <Plus className="w-3 h-3 mr-1" /> Add Category
   404→            </Button>
   405→          </div>
   406→
   407→          {showAdd && (
   408→            <Card className="border-0 shadow-md bg-orange-50/50">
   409→              <CardContent className="p-4 space-y-2">
   410→                <Input placeholder="Category name" value={addName} onChange={e => setAddName(e.target.value)} className="rounded-xl" />
   411→                <Input placeholder="Slug (auto-generated if empty)" value={addSlug} onChange={e => setAddSlug(e.target.value)} className="rounded-xl" />
   412→                <Input placeholder="Description" value={addDesc} onChange={e => setAddDesc(e.target.value)} className="rounded-xl" />
   413→                <ImageUploadSection imageUrl={addImageUrl} onImageChange={setAddImageUrl} label="Category Image (optional)" />
   414→                <div className="flex gap-2">
   415→                  <Button size="sm" className="rounded-xl" onClick={handleAddCategory} disabled={!addName.trim()}>Add</Button>
   416→                  <Button size="sm" variant="outline" className="rounded-xl" onClick={() => setShowAdd(false)}>Cancel</Button>
   417→                </div>
   418→              </CardContent>
   419→            </Card>
   420→          )}
   421→
   422→          {categories.length === 0 ? (
   423→            <Card className="border-0 shadow-sm">
   424→              <CardContent className="p-8 text-center">
   425→                <BookOpen className="w-12 h-12 text-gray-300 mx-auto mb-3" />
   426→                <p className="text-gray-400 text-sm">No categories found</p>
   427→              </CardContent>
   428→            </Card>
   429→          ) : (
   430→            categories.map(cat => (
   431→              <Card key={cat.id} className="border-0 shadow-sm hover:shadow-md transition-shadow">
   432→                <CardContent className="p-3 flex items-center gap-3">
   433→                  {editingId === cat.id ? (
   434→                    <div className="flex-1 space-y-2">
   435→                      <Input value={editName} onChange={e => setEditName(e.target.value)} className="rounded-xl text-sm" />
   436→                      <Input value={editDesc} onChange={e => setEditDesc(e.target.value)} placeholder="Description" className="rounded-xl text-sm" />
   437→                      <ImageUploadSection imageUrl={editImageUrl} onImageChange={setEditImageUrl} label="Image" />
   438→                      <div className="flex gap-2">
   439→                        <Button size="sm" className="rounded-xl" onClick={() => handleSaveEdit('category', cat.id)}>
   440→                          <Save className="w-3 h-3 mr-1" /> Save
   441→                        </Button>
   442→                        <Button size="sm" variant="outline" className="rounded-xl" onClick={() => setEditingId(null)}>Cancel</Button>
   443→                      </div>
   444→                    </div>
   445→                  ) : (
   446→                    <>
   447→                      <div className="w-10 h-10 rounded-xl bg-gradient-to-br from-orange-100 to-orange-200 flex items-center justify-center flex-shrink-0 overflow-hidden">
   448→                        {cat.imageUrl ? (
   449→                          <img src={cat.imageUrl} alt={cat.name} className="w-full h-full object-cover" />
   450→                        ) : (
   451→                          <span className="text-lg">{cat.icon || '📋'}</span>
   452→                        )}
   453→                      </div>
   454→                      <button className="flex-1 min-w-0 text-left" onClick={() => handleSelectCategory(cat)}>
   455→                        <p className="font-semibold text-sm truncate">{cat.name}</p>
   456→                        <p className="text-gray-400 text-xs truncate">{cat.description || cat.slug}</p>
   457→                        {cat.imageUrl && <Badge className="text-[8px] mt-0.5 bg-blue-100 text-blue-700">Image</Badge>}
   458→                      </button>
   459→                      <div className="flex items-center gap-1">
   460→                        <button onClick={() => handleStartEdit(cat.id, cat.name, cat.description, cat.imageUrl)} className="w-8 h-8 rounded-lg bg-blue-50 hover:bg-blue-100 flex items-center justify-center">
   461→                          <Edit3 className="w-3.5 h-3.5 text-blue-500" />
   462→                        </button>
   463→                        <button onClick={() => handleDelete('category', cat.id)} className="w-8 h-8 rounded-lg bg-red-50 hover:bg-red-100 flex items-center justify-center">
   464→                          <Trash2 className="w-3.5 h-3.5 text-red-400" />
   465→                        </button>
   466→                        <button onClick={() => handleSelectCategory(cat)} className="w-8 h-8 rounded-lg bg-gray-50 hover:bg-gray-100 flex items-center justify-center">
   467→                          <ChevronRight className="w-4 h-4 text-gray-400" />
   468→                        </button>
   469→                      </div>
   470→                    </>
   471→                  )}
   472→                </CardContent>
   473→              </Card>
   474→            ))
   475→          )}
   476→        </div>
   477→      )}
   478→
   479→      {/* ===== EXAMS VIEW ===== */}
   480→      {!loading && viewLevel === 'exams' && (
   481→        <div className="space-y-4">
   482→          <div className="flex items-center justify-between">
   483→            <h3 className="font-bold text-sm">Exams in {selectedCategoryName} ({exams.length})</h3>
   484→            <Button size="sm" className="rounded-xl text-xs" onClick={() => { setShowAdd(!showAdd); setAddName(''); setAddDesc(''); setAddSlug(''); setAddImageUrl('') }}>
   485→              <Plus className="w-3 h-3 mr-1" /> Add Exam
   486→            </Button>
   487→          </div>
   488→
   489→          {showAdd && (
   490→            <Card className="border-0 shadow-md bg-orange-50/50">
   491→              <CardContent className="p-4 space-y-2">
   492→                <Input placeholder="Exam name" value={addName} onChange={e => setAddName(e.target.value)} className="rounded-xl" />
