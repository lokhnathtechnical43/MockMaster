   220→// ============================================================
   221→// Helper: Safe Firestore call with fallback
   222→// ============================================================
   223→
   224→async function firestoreOperation<T>(
   225→  firestoreFn: () => Promise<T>,
   226→  fallbackFn: () => T
   227→): Promise<T> {
   228→  if (!useFirestore || !isFirebaseReady() || !db) {
   229→    return fallbackFn()
   230→  }
   231→  try {
   232→    return await firestoreFn()
   233→  } catch (error) {
   234→    console.error('[Firestore] Operation failed, falling back to local data:', error)
   235→    return fallbackFn()
   236→  }
   237→}

   243→/**
   244→ * Get all categories with their nested exams.
   245→ * Firestore: fetches categories & exams separately, then merges.
   246→ * Fallback: returns local-data categories.
   247→ */
   248→export async function getCategories(): Promise<LocalExamCategory[]> {
   249→  return firestoreOperation(
   250→    async () => {
   251→      const catSnap = await getDocs(
   252→        query(collection(db, COLLECTIONS.categories), orderBy('order'))
   253→      )
   254→      const examSnap = await getDocs(
   255→        query(collection(db, COLLECTIONS.exams), orderBy('order'))
   256→      )
   257→
   258→      const examsByCategory: Record<string, FirestoreExam[]> = {}
   259→      examSnap.docs.forEach((d) => {
   260→        const exam = { id: d.id, ...d.data() } as FirestoreExam
   261→        if (!examsByCategory[exam.categoryId]) examsByCategory[exam.categoryId] = []
   262→        examsByCategory[exam.categoryId].push(exam)
   263→      })
   264→
   265→      return catSnap.docs.map((d) => {
   266→        const cat = { id: d.id, ...d.data() } as FirestoreExamCategory
   267→        const catExams = examsByCategory[cat.id] || []
   268→        return {
   269→          id: cat.id,
   270→          name: cat.name,
   271→          slug: cat.slug,
   272→          icon: cat.icon,
   273→          description: cat.description,
   274→          order: cat.order,
   275→          exams: catExams.map((e) => ({
   276→            id: e.id,
   277→            name: e.name,
   278→            slug: e.slug,
   279→            description: e.description,
   280→            totalQuestions: e.totalQuestions,
   281→            duration: e.duration,
   282→            markingScheme: e.markingScheme,
   283→            order: e.order,
   284→            testCount: e.testCount,
   285→          })),
   286→        } as LocalExamCategory
   287→      })
   288→    },
   289→    () => getLocalCategories()
   290→  )
   291→}

   346→/**
   347→ * Get exams, optionally filtered by category.
   348→ */
   349→export async function getExams(categoryId?: string): Promise<LocalExam[]> {
   350→  return firestoreOperation(
   351→    async () => {
   352→      const constraints: QueryConstraint[] = [orderBy('order')]
   353→      if (categoryId) constraints.push(where('categoryId', '==', categoryId))
   354→
   355→      const snap = await getDocs(query(collection(db, COLLECTIONS.exams), ...constraints))
   356→      return snap.docs.map((d) => {
   357→        const e = { id: d.id, ...d.data() } as FirestoreExam
   358→        return {
   359→          id: e.id,
   360→          name: e.name,
   361→          slug: e.slug,
   362→          description: e.description,
   363→          totalQuestions: e.totalQuestions,
   364→          duration: e.duration,
   365→          markingScheme: e.markingScheme,
   366→          order: e.order,
   367→          testCount: e.testCount,
   368→        } as LocalExam
   369→      })
   370→    },
   371→    () => {
   372→      const cats = getLocalCategories()
   373→      const allExams = cats.flatMap((c) => c.exams)
   374→      if (categoryId) return allExams // Local data has exams nested; filter is approximate
   375→      return allExams
   376→    }
   377→  )
   378→}

   410→/**
   411→ * Delete an exam and all its associated tests and questions.
   412→ */
   413→export async function deleteExam(id: string): Promise<void> {
   414→  try {
   415→    // Delete associated tests
   416→    const testSnap = await getDocs(
   417→      query(collection(db, COLLECTIONS.tests), where('examId', '==', id))
   418→    )
   419→    const batch = writeBatch(db)
   420→    for (const testDoc of testSnap.docs) {
   421→      // Delete associated questions
   422→      const qSnap = await getDocs(
   423→        query(collection(db, COLLECTIONS.questions), where('testId', '==', testDoc.id))
   424→      )
   425→      qSnap.docs.forEach((q) => batch.delete(q.ref))
   426→      batch.delete(testDoc.ref)
   427→    }
   428→    batch.delete(doc(db, COLLECTIONS.exams, id))
   429→    await batch.commit()
   430→  } catch (error) {
   431→    console.error('[Firestore] deleteExam error:', error)
   432→    throw error
   433→  }
   434→}

   477→/**
   478→ * Get a single test by ID, including its questions.
   479→ */
   480→export async function getTestById(id: string): Promise<LocalTest | null> {
   481→  return firestoreOperation(
   482→    async () => {
   483→      const testDoc = await getDoc(doc(db, COLLECTIONS.tests, id))
   484→      if (!testDoc.exists()) return null
   485→
   486→      const t = { id: testDoc.id, ...testDoc.data() } as FirestoreTest
   487→
   488→      // Fetch questions for this test
   489→      const qSnap = await getDocs(
   490→        query(collection(db, COLLECTIONS.questions), where('testId', '==', id), orderBy('order'))
   491→      )
   492→      const questions: LocalQuestion[] = qSnap.docs.map((qd) => {
   493→        const q = { id: qd.id, ...qd.data() } as FirestoreQuestion
   494→        return {
   495→          id: q.id,
   496→          questionText: q.questionText,
   497→          questionImage: q.questionImage,
   498→          optionA: q.optionA,
   499→          optionB: q.optionB,
   500→          optionC: q.optionC,
   501→          optionD: q.optionD,
   502→          correctAnswer: q.correctAnswer,
   503→          explanation: q.explanation,
   504→          subject: q.subject,
   505→          order: q.order,
   506→        }
   507→      })
   508→
   509→      return {
   510→        id: t.id,
   511→        title: t.title,
   512→        slug: t.slug,
   513→        description: t.description,
   514→        totalQuestions: t.totalQuestions,
   515→        duration: t.duration,
   516→        markingCorrect: t.markingCorrect,
   517→        markingWrong: t.markingWrong,
   518→        markingSkipped: t.markingSkipped,
   519→        difficulty: t.difficulty,
   520→        isFree: t.isFree,
   521→        isLive: t.isLive,
   522→        exam: { id: t.examId, name: t.examName, slug: t.examSlug },
   523→        questions,
   524→      } as LocalTest
   525→    },
   526→    () => getLocalTestById(id) ?? null
   527→  )
   528→}

   661→/**
   662→ * Add multiple questions at once using a Firestore batch write.
   663→ * Returns the list of created questions with their generated IDs.
   664→ */
   665→export async function addBatchQuestions(
   666→  testId: string,
   667→  questions: Omit<LocalQuestion, 'id'>[]
   668→): Promise<LocalQuestion[]> {
   669→  try {
   670→    const batch = writeBatch(db)
   671→    const created: LocalQuestion[] = []
   672→
   673→    for (const qData of questions) {
   674→      const docRef = doc(collection(db, COLLECTIONS.questions))
   675→      const firestoreQ: FirestoreQuestion = {
   676→        id: docRef.id,
   677→        questionText: qData.questionText,
   678→        questionImage: qData.questionImage,
   679→        optionA: qData.optionA,
   680→        optionB: qData.optionB,
   681→        optionC: qData.optionC,
   682→        optionD: qData.optionD,
   683→        correctAnswer: qData.correctAnswer,

  1120→/**
  1121→ * Get daily stats breakdown for the last N days.
  1122→ * Groups results by date and computes daily averages.
  1123→ */
  1124→export async function getDailyStats(days: number = 7): Promise<DailyStats[]> {
  1125→  try {
  1126→    const now = new Date()
  1127→    const since = new Date(now.getTime() - days * 24 * 60 * 60 * 1000)
  1128→
  1129→    const resultsSnap = await getDocs(
  1130→      query(
  1131→        collection(db, COLLECTIONS.results),
  1132→        orderBy('createdAt', 'desc'),
  1133→        limit(500)
  1134→      )
  1135→    )
  1136→
  1137→    // Group by date
  1138→    const dailyMap: Record<
  1139→      string,
  1140→      { results: number; totalScore: number; users: Set<string> }
  1141→    > = {}
  1142→
  1143→    // Initialize all days with zero
  1144→    for (let i = 0; i < days; i++) {
  1145→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1146→      const key = d.toISOString().split('T')[0]
  1147→      dailyMap[key] = { results: 0, totalScore: 0, users: new Set() }
  1148→    }
  1149→
  1150→    resultsSnap.docs.forEach((d) => {
  1151→      const r = d.data() as FirestoreTestResult
  1152→      const dateStr =
  1153→        r.createdAt instanceof Timestamp
  1154→          ? r.createdAt.toDate().toISOString().split('T')[0]
  1155→          : String(r.createdAt).split('T')[0]
  1156→
  1157→      if (dailyMap[dateStr]) {
  1158→        dailyMap[dateStr].results++
  1159→        dailyMap[dateStr].totalScore += r.score
  1160→        dailyMap[dateStr].users.add(r.userId)
  1161→      }
  1162→    })
  1163→
  1164→    return Object.entries(dailyMap)
  1165→      .sort(([a], [b]) => a.localeCompare(b))
  1166→      .map(([date, stats]) => ({
  1167→        date,
  1168→        results: stats.results,
  1169→        avgScore: stats.results > 0 ? Math.round(stats.totalScore / stats.results) : 0,
  1170→        uniqueUsers: stats.users.size,
  1171→      }))
  1172→  } catch (error) {
  1173→    console.error('[Firestore] getDailyStats error:', error)
  1174→    // Return empty daily stats
  1175→    const stats: DailyStats[] = []
  1176→    const now = new Date()
  1177→    for (let i = days - 1; i >= 0; i--) {
  1178→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1179→      stats.push({
  1180→        date: d.toISOString().split('T')[0],
  1181→        results: 0,
  1182→        avgScore: 0,
  1183→        uniqueUsers: 0,
  1184→      })
  1185→    }
  1186→    return stats
  1187→  }
  1188→}
