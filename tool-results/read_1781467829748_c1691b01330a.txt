   221→// ============================================================
   222→// Helper: Safe Firestore call with fallback
   223→// ============================================================
   224→
   225→async function firestoreOperation<T>(
   226→  firestoreFn: () => Promise<T>,
   227→  fallbackFn: () => T
   228→): Promise<T> {
   229→  if (!useFirestore || !isFirebaseReady() || !db) {
   230→    return fallbackFn()
   231→  }
   232→  try {
   233→    return await firestoreFn()
   234→  } catch (error) {
   235→    console.error('[Firestore] Operation failed, falling back to local data:', error)
   236→    return fallbackFn()
   237→  }
   238→}

   244→/**
   245→ * Get all categories with their nested exams.
   246→ * Firestore: fetches categories & exams separately, then merges.
   247→ * Fallback: returns local-data categories.
   248→ */
   249→export async function getCategories(): Promise<LocalExamCategory[]> {
   250→  return firestoreOperation(
   251→    async () => {
   252→      const catSnap = await getDocs(
   253→        query(collection(db, COLLECTIONS.categories), orderBy('order'))
   254→      )
   255→      const examSnap = await getDocs(
   256→        query(collection(db, COLLECTIONS.exams), orderBy('order'))
   257→      )
   258→
   259→      const examsByCategory: Record<string, FirestoreExam[]> = {}
   260→      examSnap.docs.forEach((d) => {
   261→        const exam = { id: d.id, ...d.data() } as FirestoreExam
   262→        if (!examsByCategory[exam.categoryId]) examsByCategory[exam.categoryId] = []
   263→        examsByCategory[exam.categoryId].push(exam)
   264→      })
   265→
   266→      return catSnap.docs.map((d) => {
   267→        const cat = { id: d.id, ...d.data() } as FirestoreExamCategory
   268→        const catExams = examsByCategory[cat.id] || []
   269→        return {
   270→          id: cat.id,
   271→          name: cat.name,
   272→          slug: cat.slug,
   273→          icon: cat.icon,
   274→          description: cat.description,
   275→          order: cat.order,
   276→          exams: catExams.map((e) => ({
   277→            id: e.id,
   278→            name: e.name,
   279→            slug: e.slug,
   280→            description: e.description,
   281→            totalQuestions: e.totalQuestions,
   282→            duration: e.duration,
   283→            markingScheme: e.markingScheme,
   284→            order: e.order,
   285→            testCount: e.testCount,
   286→          })),
   287→        } as LocalExamCategory
   288→      })
   289→    },
   290→    () => getLocalCategories()
   291→  )
   292→}

   347→/**
   348→ * Get exams, optionally filtered by category.
   349→ */
   350→export async function getExams(categoryId?: string): Promise<LocalExam[]> {
   351→  return firestoreOperation(
   352→    async () => {
   353→      const constraints: QueryConstraint[] = [orderBy('order')]
   354→      if (categoryId) constraints.push(where('categoryId', '==', categoryId))
   355→
   356→      const snap = await getDocs(query(collection(db, COLLECTIONS.exams), ...constraints))
   357→      return snap.docs.map((d) => {
   358→        const e = { id: d.id, ...d.data() } as FirestoreExam
   359→        return {
   360→          id: e.id,
   361→          name: e.name,
   362→          slug: e.slug,
   363→          description: e.description,
   364→          totalQuestions: e.totalQuestions,
   365→          duration: e.duration,
   366→          markingScheme: e.markingScheme,
   367→          order: e.order,
   368→          testCount: e.testCount,
   369→        } as LocalExam
   370→      })
   371→    },
   372→    () => {
   373→      const cats = getLocalCategories()
   374→      const allExams = cats.flatMap((c) => c.exams)
   375→      if (categoryId) {
   376→        // Filter by category: match category id from parent
   377→        const cat = cats.find(c => c.id === categoryId)
   378→        return cat ? cat.exams : []
   379→      }
   380→      return allExams
   381→    }
   382→  )
   383→}

   415→/**
   416→ * Delete an exam and all its associated tests and questions.
   417→ */
   418→export async function deleteExam(id: string): Promise<void> {
   419→  try {
   420→    // Delete associated tests
   421→    const testSnap = await getDocs(
   422→      query(collection(db, COLLECTIONS.tests), where('examId', '==', id))
   423→    )
   424→    const batch = writeBatch(db)
   425→    for (const testDoc of testSnap.docs) {
   426→      // Delete associated questions
   427→      const qSnap = await getDocs(
   428→        query(collection(db, COLLECTIONS.questions), where('testId', '==', testDoc.id))
   429→      )
   430→      qSnap.docs.forEach((q) => batch.delete(q.ref))
   431→      batch.delete(testDoc.ref)
   432→    }
   433→    batch.delete(doc(db, COLLECTIONS.exams, id))
   434→    await batch.commit()
   435→  } catch (error) {
   436→    console.error('[Firestore] deleteExam error:', error)
   437→    throw error
   438→  }
   439→}

   482→/**
   483→ * Get a single test by ID, including its questions.

   485→export async function getTestById(id: string): Promise<LocalTest | null> {
   486→  return firestoreOperation(
   487→    async () => {
   488→      const testDoc = await getDoc(doc(db, COLLECTIONS.tests, id))
   489→      if (!testDoc.exists()) return null
   490→
   491→      const t = { id: testDoc.id, ...testDoc.data() } as FirestoreTest
   492→
   493→      // Fetch questions for this test
   494→      const qSnap = await getDocs(
   495→        query(collection(db, COLLECTIONS.questions), where('testId', '==', id), orderBy('order'))
   496→      )
   497→      const questions: LocalQuestion[] = qSnap.docs.map((qd) => {
   498→        const q = { id: qd.id, ...qd.data() } as FirestoreQuestion
   499→        return {
   500→          id: q.id,
   501→          questionText: q.questionText,
   502→          questionImage: q.questionImage,
   503→          optionA: q.optionA,
   504→          optionB: q.optionB,
   505→          optionC: q.optionC,
   506→          optionD: q.optionD,
   507→          correctAnswer: q.correctAnswer,
   508→          explanation: q.explanation,
   509→          subject: q.subject,
   510→          order: q.order,
   511→        }
   512→      })
   513→
   514→      return {
   515→        id: t.id,
   516→        title: t.title,
   517→        slug: t.slug,
   518→        description: t.description,
   519→        totalQuestions: t.totalQuestions,
   520→        duration: t.duration,
   521→        markingCorrect: t.markingCorrect,
   522→        markingWrong: t.markingWrong,
   523→        markingSkipped: t.markingSkipped,
   524→        difficulty: t.difficulty,
   525→        isFree: t.isFree,
   526→        isLive: t.isLive,
   527→        exam: { id: t.examId, name: t.examName, slug: t.examSlug },
   528→        questions,
   529→      } as LocalTest
   530→    },
   531→    () => getLocalTestById(id) ?? null
   532→  )
   533→}

   666→/**
   667→ * Add multiple questions at once using a Firestore batch write.
   668→ * Returns the list of created questions with their generated IDs.
   669→ */
   670→export async function addBatchQuestions(
   671→  testId: string,
   672→  questions: Omit<LocalQuestion, 'id'>[]
   673→): Promise<LocalQuestion[]> {
   674→  try {
   675→    const batch = writeBatch(db)
   676→    const created: LocalQuestion[] = []
   677→
   678→    for (const qData of questions) {
   679→      const docRef = doc(collection(db, COLLECTIONS.questions))
   680→      const firestoreQ: FirestoreQuestion = {
   681→        id: docRef.id,
   682→        questionText: qData.questionText,
   683→        questionImage: qData.questionImage ?? null,
   684→        optionA: qData.optionA,
   685→        optionB: qData.optionB,

  1136→/**
  1137→ * Get daily stats breakdown for the last N days.
  1138→ * Groups results by date and computes daily averages.
  1139→ */
  1140→export async function getDailyStats(days: number = 7): Promise<DailyStats[]> {
  1141→  try {
  1142→    const now = new Date()
  1143→    const since = new Date(now.getTime() - days * 24 * 60 * 60 * 1000)
  1144→
  1145→    const resultsSnap = await getDocs(
  1146→      query(
  1147→        collection(db, COLLECTIONS.results),
  1148→        orderBy('createdAt', 'desc'),
  1149→        limit(500)
  1150→      )
  1151→    )
  1152→
  1153→    // Group by date
  1154→    const dailyMap: Record<
  1155→      string,
  1156→      { results: number; totalScore: number; users: Set<string> }
  1157→    > = {}
  1158→
  1159→    // Initialize all days with zero
  1160→    for (let i = 0; i < days; i++) {
  1161→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1162→      const key = d.toISOString().split('T')[0]
  1163→      dailyMap[key] = { results: 0, totalScore: 0, users: new Set() }
  1164→    }
  1165→
  1166→    resultsSnap.docs.forEach((d) => {
  1167→      const r = d.data() as FirestoreTestResult
  1168→      const dateStr =
  1169→        r.createdAt instanceof Timestamp
  1170→          ? r.createdAt.toDate().toISOString().split('T')[0]
  1171→          : String(r.createdAt).split('T')[0]
  1172→
  1173→      if (dailyMap[dateStr]) {
  1174→        dailyMap[dateStr].results++
  1175→        dailyMap[dateStr].totalScore += r.score
  1176→        dailyMap[dateStr].users.add(r.userId)
  1177→      }
  1178→    })
  1179→
  1180→    return Object.entries(dailyMap)
  1181→      .sort(([a], [b]) => a.localeCompare(b))
  1182→      .map(([date, stats]) => ({
  1183→        date,
  1184→        results: stats.results,
  1185→        avgScore: stats.results > 0 ? Math.round(stats.totalScore / stats.results) : 0,
  1186→        uniqueUsers: stats.users.size,
  1187→      }))
  1188→  } catch (error) {
  1189→    console.error('[Firestore] getDailyStats error:', error)
  1190→    // Return empty daily stats
  1191→    const stats: DailyStats[] = []
  1192→    const now = new Date()
  1193→    for (let i = days - 1; i >= 0; i--) {
  1194→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1195→      stats.push({
  1196→        date: d.toISOString().split('T')[0],
  1197→        results: 0,
  1198→        avgScore: 0,
  1199→        uniqueUsers: 0,
  1200→      })
  1201→    }
  1202→    return stats
  1203→  }
  1204→}

  1214→export async function seedFirestoreIfEmpty(): Promise<boolean> {

  1217→    return false
  1218→  }

  1226→

  1229→        await batch.commit()
  1230→        batch = writeBatch(db)
  1231→        opCount = 0
  1232→      }
  1233→    }
  1234→

  1247→          totalQuestions: exam.totalQuestions, duration: exam.duration,
  1248→          markingScheme: exam.markingScheme, order: exam.order,
  1249→          testCount: exam.testCount, categoryId: catRef.id,
  1250→        })
  1251→        opCount++
  1252→      }
  1253→      await commitIfNeeded()
  1254→    }
  1255→
  1256→    for (const test of ALL_TESTS) {
  1257→      const testRef = doc(collection(db, COLLECTIONS.tests))
  1258→      batch.set(testRef, {
  1259→        title: test.title, slug: test.slug, description: test.description,
  1260→        totalQuestions: test.totalQuestions, duration: test.duration,
  1261→        markingCorrect: test.markingCorrect, markingWrong: test.markingWrong,
  1262→        markingSkipped: test.markingSkipped, difficulty: test.difficulty,
  1263→        isFree: test.isFree, isLive: test.isLive,
  1264→        examId: test.exam.id, examName: test.exam.name, examSlug: test.exam.slug,
  1265→      })
  1266→      opCount++
  1267→
  1268→      for (const q of test.questions) {
  1269→        const qRef = doc(collection(db, COLLECTIONS.questions))
  1270→        batch.set(qRef, {
  1271→          questionText: q.questionText, questionImage: q.questionImage,
  1272→          optionA: q.optionA, optionB: q.optionB, optionC: q.optionC, optionD: q.optionD,
  1273→          correctAnswer: q.correctAnswer, explanation: q.explanation,
  1274→          subject: q.subject, order: q.order, testId: testRef.id,
  1275→        })
  1276→        opCount++
  1277→      }
  1278→      await commitIfNeeded()
  1279→    }
  1280→
  1281→    // Seed default announcements
  1282→    for (const ann of DEFAULT_ANNOUNCEMENTS) {
  1283→      const annRef = doc(collection(db, COLLECTIONS.announcements))
  1284→      batch.set(annRef, { ...ann, id: annRef.id })
  1285→      opCount++
  1286→      await commitIfNeeded()
  1287→    }
  1288→
  1289→    // Seed default notifications
  1290→    for (const notif of DEFAULT_NOTIFICATIONS) {
  1291→      const notifRef = doc(collection(db, COLLECTIONS.notifications))
  1292→      batch.set(notifRef, { ...notif, id: notifRef.id })
  1293→      opCount++
  1294→      await commitIfNeeded()
  1295→    }
  1296→
  1297→    // Seed default daily tips
  1298→    for (const tip of DEFAULT_DAILY_TIPS) {
  1299→      const tipRef = doc(collection(db, COLLECTIONS.dailyTips))
  1300→      batch.set(tipRef, { ...tip, id: tipRef.id })
  1301→      opCount++
  1302→      await commitIfNeeded()
  1303→    }
  1304→
  1305→    // Commit any remaining operations
  1306→    if (opCount > 0) await batch.commit()
  1307→    console.log('[Firestore] Database seeded successfully')
  1308→    return true
  1309→  } catch (error) {
  1310→    console.error('[Firestore] Seed error:', error)
  1311→    return false
  1312→  }
  1313→}
