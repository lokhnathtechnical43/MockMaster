    62→function formatTime(seconds: number): string {
    63→  const m = Math.floor(seconds / 60)
    64→  const s = seconds % 60
    65→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
    66→}

   237→  // --- Test Functions ---
   238→  function startTest(test: LocalTest) {
   239→    const fullTest = getTestById(test.id)
   240→    if (!fullTest) return
   241→    setSelectedTest(fullTest)
   242→    setAnswers({})
   243→    setMarkedForReview(new Set())
   244→    setCurrentQuestionIndex(0)
   245→    setTimeLeft(fullTest.duration * 60)
   246→    setTestActive(true)
   247→    navigateTo('test-taking')
   248→  }
   249→
   250→  function handleFinishTest() {
   251→    if (timerRef.current) clearInterval(timerRef.current)
   252→    setTestActive(false)
   253→
   254→    if (!selectedTest) return
   255→
   256→    const test = selectedTest
   257→    const totalQuestions = test.questions.length
   258→    let correctCount = 0
   259→    let wrongCount = 0
   260→    let skippedCount = 0
   261→
   262→    test.questions.forEach(q => {
   263→      const userAnswer = answers[q.id]
   264→      if (!userAnswer) {
   265→        skippedCount++
   266→      } else if (userAnswer === q.correctAnswer) {
   267→        correctCount++
   268→      } else {
   269→        wrongCount++
   270→      }
   271→    })
   272→
   273→    const score = correctCount * test.markingCorrect - wrongCount * Math.abs(test.markingWrong)
   274→    const maxScore = totalQuestions * test.markingCorrect
   275→    const timeTaken = test.duration * 60 - timeLeft
   276→
   277→    const result = saveResult({
   278→      testId: test.id,
   279→      testName: test.title,
   280→      examName: test.exam.name,
   281→      userId: auth.getUserId() || 'anonymous',
   282→      correctCount,
   283→      wrongCount,
   284→      skippedCount,
   285→      score,
   286→      maxScore,
   287→      timeTaken,
   288→      totalQuestions,
   289→      answers,
   290→    })
   291→
   292→    setLastResult(result)
   293→    navigateTo('results')
   294→  }
   295→

   308→  function toggleReview(questionId: string) {

   313→      return next
   314→    })

   317→  // --- Stats from localStorage ---
   318→  function getUserStats() {
   319→    try {
   320→      const data = localStorage.getItem('examprep_results')
   321→      if (!data) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   322→      const results: TestResult[] = JSON.parse(data)
   323→      const userId = auth.getUserId()
   324→      const userResults = userId ? results.filter((r: TestResult) => r.userId === userId) : results
   325→      const testsTaken = userResults.length
   326→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.score / r.maxScore) * 100, 0) / testsTaken) : 0
   327→      const bestRank = 1 // simplified
   328→      return { testsTaken, avgScore, bestRank }
   329→    } catch {
   330→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   331→    }
   332→  }
