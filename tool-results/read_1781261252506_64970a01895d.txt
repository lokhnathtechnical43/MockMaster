   183→   183→
   184→   184→  function handleFinishTest() {
   185→   185→    if (timerRef.current) clearInterval(timerRef.current)
   186→   186→    setTestActive(false)
   187→   187→
   188→   188→    if (!selectedTest) return
   189→   189→

   191→   191→    const totalQuestions = test.questions.length
   192→   192→    let correctCount = 0
   193→   193→    let wrongCount = 0
   194→   194→    let skippedCount = 0
   195→   195→
   196→   196→    test.questions.forEach(q => {
   197→   197→      const userAnswer = answers[q.id]
   198→   198→      if (!userAnswer) {
   199→   199→        skippedCount++
   200→   200→      } else if (userAnswer === q.correctAnswer) {
   201→   201→        correctCount++
   202→   202→      } else {
   203→   203→        wrongCount++
   204→   204→      }
   205→   205→    })
   206→   206→
   207→   207→    const score = correctCount * test.markingCorrect - wrongCount * Math.abs(test.markingWrong)
   208→   208→    const maxScore = totalQuestions * test.markingCorrect
   209→   209→    const timeTaken = test.duration * 60 - timeLeft
   210→   210→
   211→   211→    const result = saveResult({
   212→   212→      testId: test.id,
   213→   213→      testName: test.title,
   214→   214→      examName: test.exam.name,
   215→   215→      userId: auth.getUserId() || 'anonymous',
   216→   216→      correctCount,
   217→   217→      wrongCount,
   218→   218→      skippedCount,
   219→   219→      score,
   220→   220→      maxScore,
   221→   221→      timeTaken,
   222→   222→      totalQuestions,
   223→   223→      answers,
   224→   224→    })
   225→   225→
   226→   226→    setLastResult(result)
   227→   227→    navigateTo('results')
   228→   228→  }

   263→   263→    } catch {
