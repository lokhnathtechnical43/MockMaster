    98→function formatTime(seconds: number): string {
    99→  const m = Math.floor(seconds / 60)
   100→  const s = seconds % 60
   101→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
   102→}

   159→  function toggleBookmark(questionId: string, testId?: string, testName?: string) {
   160→    setBookmarkedQuestions(prev => {
   161→      const next = new Set(prev)
   162→      if (next.has(questionId)) {
   163→        next.delete(questionId)
   164→        // Also remove context
   165→        setBookmarkContext(prevCtx => {
   166→          const nextCtx = { ...prevCtx }
   167→          delete nextCtx[questionId]
   168→          try { localStorage.setItem('examprep_bookmark_context', JSON.stringify(nextCtx)) } catch {}
   169→          return nextCtx
   170→        })
   171→      } else {
   172→        next.add(questionId)
   173→        // Also save context if provided
   174→        if (testId && testName) {
   175→          setBookmarkContext(prevCtx => {
   176→            const nextCtx = { ...prevCtx, [questionId]: { testId, testName } }
   177→            try { localStorage.setItem('examprep_bookmark_context', JSON.stringify(nextCtx)) } catch {}
   178→            return nextCtx
   179→          })
   180→        }
   181→      }
   182→      try { localStorage.setItem('examprep_bookmarks', JSON.stringify([...next])) } catch {}
   183→      return next
   184→    })
   185→  }

   662→  // --- Stats from localStorage ---
   663→  function getUserStats() {
   664→    try {
   665→      // Use getLocalResults() which reads from the correct 'mockmaster_results' key
   666→      const allData: TestResult[] = getLocalResults()
   667→      if (allData.length === 0) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   668→      const userId = auth.getUserId()
   669→      const userResults = userId ? allData.filter((r: TestResult) => r.userId === userId) : allData
   670→      const testsTaken = userResults.length
   671→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.maxScore > 0 ? (r.score / r.maxScore) * 100 : 0), 0) / testsTaken) : 0
   672→      // Calculate best rank from leaderboard data across all tests taken by user
   673→      let bestRank = 0
   674→      if (testsTaken > 0) {
   675→        const uniqueTests = [...new Set(userResults.map(r => r.testId))]
   676→        uniqueTests.forEach(testId => {
   677→          const testResults = allData
   678→            .filter(r => r.testId === testId)
   679→            .sort((a, b) => b.score - a.score || a.timeTaken - b.timeTaken)
   680→          const rank = testResults.findIndex(r => r.userId === userId) + 1
   681→          if (rank > 0 && (bestRank === 0 || rank < bestRank)) bestRank = rank
   682→        })
   683→      }
   684→      return { testsTaken, avgScore, bestRank }
   685→    } catch {
   686→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   687→    }
   688→  }
