    62→/** Firestore batch operation limit (with safety margin) */
    63→const BATCH_LIMIT = 450
    64→
    65→/**
    66→ * Helper to commit a batch and start a new one when approaching the 500-op limit.
    67→ * Returns the new batch after committing if the threshold was reached.
    68→ */
    69→async function commitIfNeeded(batch: WriteBatch, opCount: number): Promise<{ batch: WriteBatch; opCount: number }> {
    70→  if (opCount >= BATCH_LIMIT) {
    71→    await batch.commit()
    72→    return { batch: writeBatch(db), opCount: 0 }
    73→  }
    74→  return { batch, opCount }
    75→}

   237→// ============================================================
   238→// Helper: Safe Firestore call with fallback
   239→// ============================================================
   240→
   241→async function firestoreOperation<T>(
   242→  firestoreFn: () => Promise<T>,
   243→  fallbackFn: () => T
   244→): Promise<T> {
   245→  if (!useFirestore || !isFirebaseReady() || !db) {
   246→    return fallbackFn()
   247→  }
   248→  try {
   249→    return await firestoreFn()
   250→  } catch (error) {
   251→    console.error('[Firestore] Operation failed, falling back to local data:', error)
   252→    return fallbackFn()
   253→  }
   254→}

   260→/**
   261→ * Get all categories with their nested exams.
   262→ * Firestore: fetches categories & exams separately, then merges.
   263→ * Fallback: returns local-data categories.
   264→ */
   265→export async function getCategories(): Promise<LocalExamCategory[]> {
   266→  return firestoreOperation(
   267→    async () => {
   268→      const catSnap = await getDocs(
   269→        query(collection(db, COLLECTIONS.categories), orderBy('order'))
   270→      )
   271→      const examSnap = await getDocs(
   272→        query(collection(db, COLLECTIONS.exams), orderBy('order'))
   273→      )
   274→
   275→      const examsByCategory: Record<string, FirestoreExam[]> = {}
   276→      examSnap.docs.forEach((d) => {
   277→        const exam = { id: d.id, ...d.data() } as FirestoreExam
   278→        if (!examsByCategory[exam.categoryId]) examsByCategory[exam.categoryId] = []
   279→        examsByCategory[exam.categoryId].push(exam)
   280→      })
   281→
   282→      return catSnap.docs.map((d) => {
   283→        const cat = { id: d.id, ...d.data() } as FirestoreExamCategory
   284→        const catExams = examsByCategory[cat.id] || []
   285→        return {
   286→          id: cat.id,
   287→          name: cat.name,
   288→          slug: cat.slug,
   289→          icon: cat.icon,
   290→          description: cat.description,
   291→          order: cat.order,
   292→          exams: catExams.map((e) => ({
   293→            id: e.id,
   294→            name: e.name,
   295→            slug: e.slug,
   296→            description: e.description,
   297→            totalQuestions: e.totalQuestions,
   298→            duration: e.duration,
   299→            markingScheme: e.markingScheme,
   300→            order: e.order,
   301→            testCount: e.testCount,
   302→          })),
   303→        } as LocalExamCategory
   304→      })
   305→    },
   306→    () => getLocalCategories()
   307→  )
   308→}

   340→/**
   341→ * Delete a category and all its associated exams.
   342→ */
   343→export async function deleteCategory(id: string): Promise<void> {
   344→  try {
   345→    // Delete associated exams first (which also deletes their tests and questions)
   346→    const examSnap = await getDocs(
   347→      query(collection(db, COLLECTIONS.exams), where('categoryId', '==', id))
   348→    )
   349→    for (const examDoc of examSnap.docs) {
   350→      await deleteExam(examDoc.id)
   351→    }
   352→    // Delete the category document itself
   353→    await deleteDoc(doc(db, COLLECTIONS.categories, id))
   354→  } catch (error) {
   355→    console.error('[Firestore] deleteCategory error:', error)
   356→    throw error
   357→  }

   364→/**
   365→ * Get exams, optionally filtered by category.
   366→ */
   367→export async function getExams(categoryId?: string): Promise<LocalExam[]> {
   368→  return firestoreOperation(
   369→    async () => {
   370→      const constraints: QueryConstraint[] = [orderBy('order')]
   371→      if (categoryId) constraints.push(where('categoryId', '==', categoryId))
   372→
   373→      const snap = await getDocs(query(collection(db, COLLECTIONS.exams), ...constraints))
   374→      return snap.docs.map((d) => {
   375→        const e = { id: d.id, ...d.data() } as FirestoreExam
   376→        return {
   377→          id: e.id,
   378→          name: e.name,
   379→          slug: e.slug,
   380→          description: e.description,
   381→          totalQuestions: e.totalQuestions,
   382→          duration: e.duration,
   383→          markingScheme: e.markingScheme,
   384→          order: e.order,
   385→          testCount: e.testCount,
   386→        } as LocalExam
   387→      })
   388→    },
   389→    () => {
   390→      const cats = getLocalCategories()
   391→      const allExams = cats.flatMap((c) => c.exams)
   392→      if (categoryId) {
   393→        // Filter by category: match category id from parent
   394→        const cat = cats.find(c => c.id === categoryId)
   395→        return cat ? cat.exams : []
   396→      }
   397→      return allExams
   398→    }
   399→  )
   400→}

   523→        }
   524→      })
   525→
   526→      return {
   527→        id: t.id,
   528→        title: t.title,
   529→        slug: t.slug,
   530→        description: t.description,
   531→        totalQuestions: t.totalQuestions,

   535→        markingSkipped: t.markingSkipped,
   536→        difficulty: t.difficulty,

   540→        questions,

   577→/**
   578→ * Delete a test and all its associated questions.
   579→ * Handles batch chunking for tests with many questions (>450).
   580→ */
   581→export async function deleteTest(id: string): Promise<void> {
   582→  try {
   583→    const qSnap = await getDocs(
   584→      query(collection(db, COLLECTIONS.questions), where('testId', '==', id))
   585→    )
   586→    const BATCH_LIMIT = 450
   587→    let batch = writeBatch(db)
   588→    let opCount = 0
   589→    qSnap.docs.forEach((q) => {
   590→      batch.delete(q.ref)
   591→      opCount++
   592→      if (opCount >= BATCH_LIMIT) {
   593→        batch.commit()
   594→        batch = writeBatch(db)
   595→        opCount = 0
   596→      }
   597→    })
   598→    batch.delete(doc(db, COLLECTIONS.tests, id))
   599→    opCount++
   600→    if (opCount > 0) await batch.commit()
   601→  } catch (error) {
   602→    console.error('[Firestore] deleteTest error:', error)
   603→    throw error
   604→  }
   605→}

   690→/**
   691→ * Add multiple questions at once using a Firestore batch write.
   692→ * Returns the list of created questions with their generated IDs.
   693→ */
   694→export async function addBatchQuestions(
   695→  testId: string,
   696→  questions: Omit<LocalQuestion, 'id'>[]
   697→): Promise<LocalQuestion[]> {
   698→  try {
   699→    const BATCH_LIMIT = 450
   700→    const created: LocalQuestion[] = []
   701→    let batch = writeBatch(db)
   702→    let opCount = 0
   703→
   704→    for (const qData of questions) {
   705→      const docRef = doc(collection(db, COLLECTIONS.questions))
   706→      const firestoreQ: FirestoreQuestion = {
   707→        id: docRef.id,
   708→        questionText: qData.questionText,
   709→        questionImage: qData.questionImage ?? null,
   710→        optionA: qData.optionA,
   711→        optionB: qData.optionB,
   712→        optionC: qData.optionC,
   713→        optionD: qData.optionD,
   714→        correctAnswer: qData.correctAnswer,
   715→        explanation: qData.explanation ?? null,
   716→        subject: qData.subject ?? null,
   717→        order: qData.order ?? 0,
   718→        testId,
   719→      }
   720→      batch.set(docRef, firestoreQ)
   721→      created.push({ ...qData, id: docRef.id })
   722→      opCount++
   723→
   724→      if (opCount >= BATCH_LIMIT) {
   725→        await batch.commit()
   726→        batch = writeBatch(db)
   727→        opCount = 0
   728→      }
   729→    }
   730→
   731→    if (opCount > 0) await batch.commit()
   732→    return created
   733→  } catch (error) {
   734→    console.error('[Firestore] addBatchQuestions error:', error)
   735→    throw error
   736→  }
   737→}

  1182→/**
  1183→ * Get daily stats breakdown for the last N days.
  1184→ * Groups results by date and computes daily averages.
  1185→ */
  1186→export async function getDailyStats(days: number = 7): Promise<DailyStats[]> {
  1187→  try {
  1188→    const now = new Date()
  1189→    const since = new Date(now.getTime() - days * 24 * 60 * 60 * 1000)
  1190→
  1191→    const resultsSnap = await getDocs(
  1192→      query(
  1193→        collection(db, COLLECTIONS.results),
  1194→        orderBy('createdAt', 'desc'),
  1195→        limit(500)
  1196→      )
  1197→    )
  1198→
  1199→    // Group by date
  1200→    const dailyMap: Record<
  1201→      string,
  1202→      { results: number; totalScorePercent: number; users: Set<string> }
  1203→    > = {}
  1204→
  1205→    // Initialize all days with zero
  1206→    for (let i = 0; i < days; i++) {
  1207→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1208→      const key = d.toISOString().split('T')[0]
  1209→      dailyMap[key] = { results: 0, totalScorePercent: 0, users: new Set() }
  1210→    }
  1211→
  1212→    resultsSnap.docs.forEach((d) => {
  1213→      const r = d.data() as FirestoreTestResult
  1214→      const dateStr =
  1215→        r.createdAt instanceof Timestamp
  1216→          ? r.createdAt.toDate().toISOString().split('T')[0]
  1217→          : String(r.createdAt).split('T')[0]
  1218→
  1219→      if (dailyMap[dateStr]) {
  1220→        dailyMap[dateStr].results++
  1221→        if (r.maxScore > 0) {
  1222→          dailyMap[dateStr].totalScorePercent += (r.score / r.maxScore) * 100
  1223→        }
  1224→        dailyMap[dateStr].users.add(r.userId)
  1225→      }
  1226→    })
  1227→
  1228→    return Object.entries(dailyMap)
  1229→      .sort(([a], [b]) => a.localeCompare(b))
  1230→      .map(([date, stats]) => ({
  1231→        date,
  1232→        results: stats.results,
  1233→        avgScore: stats.results > 0 ? Math.round(stats.totalScorePercent / stats.results) : 0,
  1234→        uniqueUsers: stats.users.size,
  1235→      }))
  1236→  } catch (error) {
  1237→    console.error('[Firestore] getDailyStats error:', error)
  1238→    // Return empty daily stats
  1239→    const stats: DailyStats[] = []
  1240→    const now = new Date()
  1241→    for (let i = days - 1; i >= 0; i--) {
  1242→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1243→      stats.push({
  1244→        date: d.toISOString().split('T')[0],
  1245→        results: 0,
  1246→        avgScore: 0,
  1247→        uniqueUsers: 0,
  1248→      })
  1249→    }
  1250→    return stats
  1251→  }
  1252→}

  1326→      await commitIfNeeded()
  1327→    }
  1328→
  1329→    // Seed default announcements
  1330→    for (const ann of DEFAULT_ANNOUNCEMENTS) {

  1335→    }
  1336→

  1350→      await commitIfNeeded()
  1351→    }
  1352→
  1353→    // Commit any remaining operations
  1354→    if (opCount > 0) await batch.commit()
  1355→    console.log('[Firestore] Database seeded successfully')
  1356→    return true
  1357→  } catch (error) {
  1358→    console.error('[Firestore] Seed error:', error)
  1359→    return false
  1360→  }
  1361→}
