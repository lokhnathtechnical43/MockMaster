   215→// ============================================================
   216→// Helper: Safe Firestore call with fallback
   217→// ============================================================
   218→
   219→async function firestoreOperation<T>(
   220→  firestoreFn: () => Promise<T>,
   221→  fallbackFn: () => T
   222→): Promise<T> {
   223→  if (!useFirestore || !isFirebaseReady() || !db) {
   224→    return fallbackFn()
   225→  }
   226→  try {
   227→    return await firestoreFn()
   228→  } catch (error) {
   229→    console.error('[Firestore] Operation failed, falling back to local data:', error)
   230→    return fallbackFn()
   231→  }
   232→}

   238→/**
   239→ * Get all categories with their nested exams.
   240→ * Firestore: fetches categories & exams separately, then merges.
   241→ * Fallback: returns local-data categories.
   242→ */
   243→export async function getCategories(): Promise<LocalExamCategory[]> {
   244→  return firestoreOperation(
   245→    async () => {
   246→      const catSnap = await getDocs(
   247→        query(collection(db, COLLECTIONS.categories), orderBy('order'))
   248→      )
   249→      const examSnap = await getDocs(
   250→        query(collection(db, COLLECTIONS.exams), orderBy('order'))
   251→      )
   252→
   253→      const examsByCategory: Record<string, FirestoreExam[]> = {}
   254→      examSnap.docs.forEach((d) => {
   255→        const exam = { id: d.id, ...d.data() } as FirestoreExam
   256→        if (!examsByCategory[exam.categoryId]) examsByCategory[exam.categoryId] = []
   257→        examsByCategory[exam.categoryId].push(exam)
   258→      })
   259→
   260→      return catSnap.docs.map((d) => {
   261→        const cat = { id: d.id, ...d.data() } as FirestoreExamCategory
   262→        const catExams = examsByCategory[cat.id] || []
   263→        return {
   264→          id: cat.id,
   265→          name: cat.name,
   266→          slug: cat.slug,
   267→          icon: cat.icon,
   268→          description: cat.description,
   269→          order: cat.order,
   270→          exams: catExams.map((e) => ({
   271→            id: e.id,
   272→            name: e.name,
   273→            slug: e.slug,
   274→            description: e.description,
   275→            totalQuestions: e.totalQuestions,
   276→            duration: e.duration,
   277→            markingScheme: e.markingScheme,
   278→            order: e.order,
   279→            testCount: e.testCount,
   280→          })),
   281→        } as LocalExamCategory
   282→      })
   283→    },
   284→    () => getLocalCategories()
   285→  )
   286→}

   341→/**
   342→ * Get exams, optionally filtered by category.
   343→ */
   344→export async function getExams(categoryId?: string): Promise<LocalExam[]> {
   345→  return firestoreOperation(
   346→    async () => {
   347→      const constraints: QueryConstraint[] = [orderBy('order')]
   348→      if (categoryId) constraints.push(where('categoryId', '==', categoryId))
   349→
   350→      const snap = await getDocs(query(collection(db, COLLECTIONS.exams), ...constraints))
   351→      return snap.docs.map((d) => {
   352→        const e = { id: d.id, ...d.data() } as FirestoreExam
   353→        return {
   354→          id: e.id,
   355→          name: e.name,
   356→          slug: e.slug,
   357→          description: e.description,
   358→          totalQuestions: e.totalQuestions,
   359→          duration: e.duration,
   360→          markingScheme: e.markingScheme,
   361→          order: e.order,
   362→          testCount: e.testCount,
   363→        } as LocalExam
   364→      })
   365→    },
   366→    () => {
   367→      const cats = getLocalCategories()
   368→      const allExams = cats.flatMap((c) => c.exams)
   369→      if (categoryId) return allExams // Local data has exams nested; filter is approximate
   370→      return allExams
   371→    }
   372→  )
   373→}

   405→/**
   406→ * Delete an exam and all its associated tests and questions.
   407→ */
   408→export async function deleteExam(id: string): Promise<void> {
   409→  try {
   410→    // Delete associated tests
   411→    const testSnap = await getDocs(
   412→      query(collection(db, COLLECTIONS.tests), where('examId', '==', id))
   413→    )
   414→    const batch = writeBatch(db)
   415→    for (const testDoc of testSnap.docs) {
   416→      // Delete associated questions
   417→      const qSnap = await getDocs(
   418→        query(collection(db, COLLECTIONS.questions), where('testId', '==', testDoc.id))
   419→      )
   420→      qSnap.docs.forEach((q) => batch.delete(q.ref))
   421→      batch.delete(testDoc.ref)
   422→    }
   423→    batch.delete(doc(db, COLLECTIONS.exams, id))
   424→    await batch.commit()
   425→  } catch (error) {
   426→    console.error('[Firestore] deleteExam error:', error)
   427→    throw error
   428→  }
   429→}

   472→/**
   473→ * Get a single test by ID, including its questions.
   474→ */
   475→export async function getTestById(id: string): Promise<LocalTest | null> {
   476→  return firestoreOperation(
   477→    async () => {
   478→      const testDoc = await getDoc(doc(db, COLLECTIONS.tests, id))
   479→      if (!testDoc.exists()) return null
   480→
   481→      const t = { id: testDoc.id, ...testDoc.data() } as FirestoreTest
   482→
   483→      // Fetch questions for this test
   484→      const qSnap = await getDocs(
   485→        query(collection(db, COLLECTIONS.questions), where('testId', '==', id), orderBy('order'))
   486→      )
   487→      const questions: LocalQuestion[] = qSnap.docs.map((qd) => {
   488→        const q = { id: qd.id, ...qd.data() } as FirestoreQuestion
   489→        return {
   490→          id: q.id,
   491→          questionText: q.questionText,
   492→          questionImage: q.questionImage,
   493→          optionA: q.optionA,
   494→          optionB: q.optionB,
   495→          optionC: q.optionC,
   496→          optionD: q.optionD,
   497→          correctAnswer: q.correctAnswer,
   498→          explanation: q.explanation,
   499→          subject: q.subject,
   500→          order: q.order,
   501→        }
   502→      })
   503→
   504→      return {
   505→        id: t.id,
   506→        title: t.title,
   507→        slug: t.slug,
   508→        description: t.description,
   509→        totalQuestions: t.totalQuestions,
   510→        duration: t.duration,
   511→        markingCorrect: t.markingCorrect,
   512→        markingWrong: t.markingWrong,
   513→        markingSkipped: t.markingSkipped,
   514→        difficulty: t.difficulty,
   515→        isFree: t.isFree,
   516→        isLive: t.isLive,
   517→        exam: { id: t.examId, name: t.examName, slug: t.examSlug },
   518→        questions,
   519→      } as LocalTest
   520→    },
   521→    () => getLocalTestById(id) ?? null
   522→  )
   523→}

   656→/**
   657→ * Add multiple questions at once using a Firestore batch write.
   658→ * Returns the list of created questions with their generated IDs.
   659→ */
   660→export async function addBatchQuestions(
   661→  testId: string,
   662→  questions: Omit<LocalQuestion, 'id'>[]
   663→): Promise<LocalQuestion[]> {
   664→  try {
   665→    const batch = writeBatch(db)
   666→    const created: LocalQuestion[] = []
   667→
   668→    for (const qData of questions) {
   669→      const docRef = doc(collection(db, COLLECTIONS.questions))
   670→      const firestoreQ: FirestoreQuestion = {
   671→        id: docRef.id,
   672→        questionText: qData.questionText,
   673→        questionImage: qData.questionImage,
   674→        optionA: qData.optionA,
   675→        optionB: qData.optionB,
   676→        optionC: qData.optionC,
   677→        optionD: qData.optionD,
   678→        correctAnswer: qData.correctAnswer,
   679→        explanation: qData.explanation,
   680→        subject: qData.subject,
   681→        order: qData.order,
   682→        testId,
   683→      }
   684→      batch.set(docRef, firestoreQ)
   685→      created.push({ ...qData, id: docRef.id })
   686→    }
   687→
   688→    await batch.commit()

  1068→/**
  1069→ * Get daily stats breakdown for the last N days.
  1070→ * Groups results by date and computes daily averages.
  1071→ */
  1072→export async function getDailyStats(days: number = 7): Promise<DailyStats[]> {
  1073→  try {
  1074→    const now = new Date()
  1075→    const since = new Date(now.getTime() - days * 24 * 60 * 60 * 1000)
  1076→
  1077→    const resultsSnap = await getDocs(
  1078→      query(
  1079→        collection(db, COLLECTIONS.results),
  1080→        orderBy('createdAt', 'desc'),
  1081→        limit(500)
  1082→      )
  1083→    )
  1084→
  1085→    // Group by date
  1086→    const dailyMap: Record<
  1087→      string,
  1088→      { results: number; totalScore: number; users: Set<string> }
  1089→    > = {}
  1090→
  1091→    // Initialize all days with zero
  1092→    for (let i = 0; i < days; i++) {
  1093→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1094→      const key = d.toISOString().split('T')[0]
  1095→      dailyMap[key] = { results: 0, totalScore: 0, users: new Set() }
  1096→    }
  1097→
  1098→    resultsSnap.docs.forEach((d) => {
  1099→      const r = d.data() as FirestoreTestResult
  1100→      const dateStr =
  1101→        r.createdAt instanceof Timestamp
  1102→          ? r.createdAt.toDate().toISOString().split('T')[0]
  1103→          : String(r.createdAt).split('T')[0]
  1104→
  1105→      if (dailyMap[dateStr]) {
  1106→        dailyMap[dateStr].results++
  1107→        dailyMap[dateStr].totalScore += r.score
  1108→        dailyMap[dateStr].users.add(r.userId)
  1109→      }
  1110→    })
  1111→
  1112→    return Object.entries(dailyMap)
  1113→      .sort(([a], [b]) => a.localeCompare(b))
  1114→      .map(([date, stats]) => ({
  1115→        date,
  1116→        results: stats.results,
  1117→        avgScore: stats.results > 0 ? Math.round(stats.totalScore / stats.results) : 0,
  1118→        uniqueUsers: stats.users.size,
  1119→      }))
  1120→  } catch (error) {
  1121→    console.error('[Firestore] getDailyStats error:', error)
  1122→    // Return empty daily stats
  1123→    const stats: DailyStats[] = []
  1124→    const now = new Date()
  1125→    for (let i = days - 1; i >= 0; i--) {
  1126→      const d = new Date(now.getTime() - i * 24 * 60 * 60 * 1000)
  1127→      stats.push({
  1128→        date: d.toISOString().split('T')[0],
  1129→        results: 0,
  1130→        avgScore: 0,
  1131→        uniqueUsers: 0,
  1132→      })
  1133→    }
  1134→    return stats
  1135→  }
  1136→}
