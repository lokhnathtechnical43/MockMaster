   399→  // --- Test Functions ---
   400→  async function startTest(test: LocalTest) {
   401→    const fullTest = await fetchTestById(test.id)
   402→    if (!fullTest) return
   403→    setSelectedTest(fullTest)
   404→    setAnswers({})
   405→    setMarkedForReview(new Set())

   407→    setTimeLeft(fullTest.duration * 60)
   408→    setTestActive(true)

   411→
   412→  async function handleFinishTest() {
   413→    if (timerRef.current) clearInterval(timerRef.current)
   414→    setTestActive(false)
   415→
   416→    if (!selectedTest) return
   417→
   418→    const test = selectedTest
   419→    const totalQuestions = test.questions.length
   420→    let correctCount = 0
   421→    let wrongCount = 0
   422→    let skippedCount = 0
   423→
   424→    test.questions.forEach(q => {
   425→      const userAnswer = answers[q.id]
   426→      if (!userAnswer) {
   427→        skippedCount++
   428→      } else if (userAnswer === q.correctAnswer) {
   429→        correctCount++
   430→      } else {
   431→        wrongCount++
   432→      }
   433→    })
   434→
   435→    const score = correctCount * test.markingCorrect - wrongCount * Math.abs(test.markingWrong)
   436→    const maxScore = totalQuestions * test.markingCorrect
   437→    const timeTaken = test.duration * 60 - timeLeft
   438→
   439→    const result = await storeResult({
   440→      testId: test.id,
   441→      testName: test.title,
   442→      examName: test.exam.name,
   443→      userId: auth.getUserId() || 'anonymous',
   444→      correctCount,
   445→      wrongCount,
   446→      skippedCount,
   447→      score,
   448→      maxScore,
   449→      timeTaken,
   450→      totalQuestions,
   451→      answers,
   452→    })
   453→
   454→    setLastResult(result)
   455→    navigateTo('results')
   456→  }

   479→  // --- Stats from localStorage ---
   480→  function getUserStats() {
   481→    try {
   482→      const data = localStorage.getItem('examprep_results')
   483→      if (!data) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   484→      const results: TestResult[] = JSON.parse(data)
   485→      const userId = auth.getUserId()
   486→      const userResults = userId ? results.filter((r: TestResult) => r.userId === userId) : results
   487→      const testsTaken = userResults.length
   488→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.score / r.maxScore) * 100, 0) / testsTaken) : 0
   489→      const bestRank = 1 // simplified
   490→      return { testsTaken, avgScore, bestRank }
   491→    } catch {
   492→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   493→    }
   494→  }
