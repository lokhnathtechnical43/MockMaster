    99→function formatTime(seconds: number): string {
   100→  const m = Math.floor(seconds / 60)
   101→  const s = seconds % 60
   102→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`
   103→}

   160→  function toggleBookmark(questionId: string, testId?: string, testName?: string) {
   161→    setBookmarkedQuestions(prev => {
   162→      const next = new Set(prev)
   163→      if (next.has(questionId)) {
   164→        next.delete(questionId)
   165→        // Also remove context
   166→        setBookmarkContext(prevCtx => {
   167→          const nextCtx = { ...prevCtx }
   168→          delete nextCtx[questionId]
   169→          try { localStorage.setItem('examprep_bookmark_context', JSON.stringify(nextCtx)) } catch {}
   170→          return nextCtx
   171→        })
   172→      } else {
   173→        next.add(questionId)
   174→        // Also save context if provided
   175→        if (testId && testName) {
   176→          setBookmarkContext(prevCtx => {
   177→            const nextCtx = { ...prevCtx, [questionId]: { testId, testName } }
   178→            try { localStorage.setItem('examprep_bookmark_context', JSON.stringify(nextCtx)) } catch {}
   179→            return nextCtx
   180→          })
   181→        }
   182→      }
   183→      try { localStorage.setItem('examprep_bookmarks', JSON.stringify([...next])) } catch {}
   184→      return next
   185→    })
   186→  }

   657→  // --- Stats from localStorage ---
   658→  function getUserStats() {
   659→    try {
   660→      // Use getLocalResults() which reads from the correct 'mockmaster_results' key
   661→      const allData: TestResult[] = getLocalResults()
   662→      if (allData.length === 0) return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   663→      const userId = auth.getUserId()
   664→      const userResults = userId ? allData.filter((r: TestResult) => r.userId === userId) : allData
   665→      const testsTaken = userResults.length
   666→      const avgScore = testsTaken > 0 ? Math.round(userResults.reduce((sum: number, r: TestResult) => sum + (r.maxScore > 0 ? (r.score / r.maxScore) * 100 : 0), 0) / testsTaken) : 0
   667→      // Calculate best rank from leaderboard data across all tests taken by user
   668→      let bestRank = 0
   669→      if (testsTaken > 0) {
   670→        const uniqueTests = [...new Set(userResults.map(r => r.testId))]
   671→        uniqueTests.forEach(testId => {
   672→          const testResults = allData
   673→            .filter(r => r.testId === testId)
   674→            .sort((a, b) => b.score - a.score || a.timeTaken - b.timeTaken)
   675→          const rank = testResults.findIndex(r => r.userId === userId) + 1
   676→          if (rank > 0 && (bestRank === 0 || rank < bestRank)) bestRank = rank
   677→        })
   678→      }
   679→      return { testsTaken, avgScore, bestRank }
   680→    } catch {
   681→      return { testsTaken: 0, avgScore: 0, bestRank: 0 }
   682→    }
   683→  }
